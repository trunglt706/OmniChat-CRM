   326→function ChannelsTab() {
   327→  const { t } = useT()
   328→  const [channels, setChannels] = useState<ChannelData[]>([])
   329→  const [loading, setLoading] = useState(true)
   330→  const [editingChannel, setEditingChannel] = useState<string | null>(null)
   331→  const [editForm, setEditForm] = useState<Record<string, string>>({})
   332→  const [saving, setSaving] = useState(false)
   333→  const [testingKey, setTestingKey] = useState<string | null>(null)
   334→  const [testResults, setTestResults] = useState<Record<string, { ok: boolean; msg: string } | null>>({})
   335→
   336→  // Load channels from API
   337→  useEffect(() => {
   338→    fetch('/api/channels')
   339→      .then(r => r.json())
   340→      .then(data => { if (Array.isArray(data)) setChannels(data) })
   341→      .catch(() => {})
   342→      .finally(() => setLoading(false))
   343→  }, [])
   344→
   345→  const toggleChannel = async (key: string) => {
   346→    const ch = channels.find(c => c.key === key)
   347→    if (!ch) return
   348→    setChannels(prev => prev.map(c => c.key === key ? { ...c, enabled: !c.enabled } : c))
   349→    await fetch('/api/channels', {
   350→      method: 'PUT',
   351→      headers: { 'Content-Type': 'application/json' },
   352→      body: JSON.stringify({ channel: key, enabled: !ch.enabled, config: ch.config }),
   353→    }).catch(() => {})
   354→  }
   355→
   356→  const startEdit = (ch: ChannelData) => {
   357→    setEditingChannel(ch.key)
   358→    setEditForm({ ...ch.config })
   359→    setTestResults(prev => ({ ...prev, [ch.key]: null }))
   360→  }
   361→
   362→  const saveChannel = async (key: string) => {
   363→    setSaving(true)
   364→    try {
   365→      const res = await fetch('/api/channels', {
   366→        method: 'PUT',
   367→        headers: { 'Content-Type': 'application/json' },
   368→        body: JSON.stringify({ channel: key, config: editForm }),
   369→      })
   370→      const data = await res.json()
   371→      setChannels(prev => prev.map(c => c.key === key ? { ...c, configured: data.configured, config: data.config || c.config } : c))
   372→      setEditingChannel(null)
   373→    } catch {}
   374→    setSaving(false)
   375→  }
   376→
   377→  const testConnection = async (key: string) => {
   378→    setTestingKey(key)
   379→    setTestResults(prev => ({ ...prev, [key]: null }))
   380→    try {
   381→      const ch = channels.find(c => c.key === key)
   382→      const res = await fetch('/api/channels/test', {
   383→        method: 'POST',
   384→        headers: { 'Content-Type': 'application/json' },
   385→        body: JSON.stringify({ channel: key, config: { ...ch?.config, ...editForm } }),
   386→      })
   387→      const result = await res.json()
   388→      setTestResults(prev => ({ ...prev, [key]: result }))
   389→      // Update channel data to reflect new test result
   390→      setChannels(prev => prev.map(c => c.key === key ? {
   391→        ...c,
   392→        lastTestAt: new Date().toISOString(),
   393→        lastTestOk: result.ok,
   394→        lastTestMsg: result.msg,
   395→      } : c))
   396→    } catch {
   397→      setTestResults(prev => ({ ...prev, [key]: { ok: false, msg: 'Lỗi kết nối đến server' } }))
   398→    }
   399→    setTestingKey(null)
   400→  }
   401→
   402→  if (loading) {
   403→    return (
   404→      <div className="flex items-center justify-center py-20">
   405→        <Loader2 className="h-5 w-5 animate-spin text-muted-foreground/40" />
   406→      </div>
   407→    )
   408→  }
   409→
   410→  return (
   411→    <div className="space-y-5 animate-fade-in">
   412→      <div>
   413→        <h3 className="text-sm font-bold">{t('channels.configTitle')}</h3>
   414→        <p className="text-xs text-muted-foreground/60 mt-1">{t('channels.configDesc')}</p>
   415→      </div>
   416→      {channels.map((ch) => {
   417→        const Icon = CHANNEL_ICONS[ch.key] || MessageSquare
   418→        const color = CHANNEL_COLORS_MAP[ch.key] || '#6b7280'
   419→        const nameKey = CHANNEL_NAME_KEYS[ch.key] || ch.key
   420→        const descKey = CHANNEL_DESC_KEYS[ch.key] || ch.key
   421→        const isEditing = editingChannel === ch.key
   422→        const testResult = testResults[ch.key]
   423→        const isTesting = testingKey === ch.key
   424→
   425→        return (
   426→          <div key={ch.key} className="glass-card rounded-2xl overflow-hidden transition-all duration-200">
   427→            <div className="p-5">
   428→              <div className="flex items-start justify-between gap-4">
   429→                <div className="flex items-start gap-3.5 min-w-0">
   430→                  <div className="h-11 w-11 rounded-xl flex items-center justify-center flex-shrink-0 shadow-sm" style={{ backgroundColor: color + '12' }}>
   431→                    <Icon className="h-5 w-5" style={{ color }} />
   432→                  </div>
   433→                  <div className="min-w-0">
   434→                    <div className="flex items-center gap-2 flex-wrap">
   435→                      <h4 className="text-[13px] font-semibold">{t(nameKey)}</h4>
   436→                      {ch.configured && ch.enabled && (
   437→                        <Badge className="text-[9px] px-1.5 py-0 h-[16px] bg-emerald-100 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-400 rounded-md font-medium">
   438→                          <Check className="h-2.5 w-2.5 mr-0.5" /> {t('channels.connected')}
   439→                        </Badge>
   440→                      )}
   441→                      {!ch.configured && ch.enabled && (
   442→                        <Badge className="text-[9px] px-1.5 py-0 h-[16px] bg-amber-100 text-amber-700 dark:bg-amber-950/40 dark:text-amber-400 rounded-md font-medium">
   443→                          {t('channels.notConfigured')}
   444→                        </Badge>
   445→                      )}
   446→                      {ch.lastTestAt && (
   447→                        <span className={cn(
   448→                          'text-[9px] font-medium',
   449→                          ch.lastTestOk ? 'text-emerald-600' : 'text-red-500'
   450→                        )}>
   451→                          {ch.lastTestOk ? t('channels.lastTestOk') : t('channels.lastTestFail')}
   452→                          {' · '}{new Date(ch.lastTestAt).toLocaleTimeString('vi-VN')}

   460→              </div>

   462→
   463→            {/* Test result banner */}
   464→            {testResult != null && (
   465→              <div className={cn(
   466→                'mx-5 mb-0 px-3.5 py-2.5 rounded-xl text-[11px] font-medium animate-slide-down',
