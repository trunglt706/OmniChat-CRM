    52→// ═══ Shared fetch cache (prevents duplicate API calls across mounts) ═══
    53→const fetchCache = new Map<string, { promise: Promise<any>; ts: number }>()
    54→const CACHE_TTL = 5000
    55→function cachedFetch(url: string): Promise<any> {
    56→  const now = Date.now()
    57→  const cached = fetchCache.get(url)
    58→  if (cached && now - cached.ts < CACHE_TTL) return cached.promise
    59→  const promise = fetch(url).then(r => r.ok ? r.json() : null)
    60→  fetchCache.set(url, { promise, ts: now })
    61→  return promise
    62→}

   163→

   165→    cachedFetch('/api/agents/me/stats')
   166→      .then(data => { if (data) setStats(data) })

   171→    if (currentUser) setForm({ name: currentUser.name, email: currentUser.email, phone: currentUser.phone, bio: currentUser.bio })

   622→// ═══ STAFF TAB ═══
   623→function StaffTab() {
   624→  const agents = useCRMStore((s) => s.agents)
   625→  const setAgents = useCRMStore((s) => s.setAgents)
   626→  const { t } = useT()
   627→  const [loading, setLoading] = useState(true)
   628→  const [showInvite, setShowInvite] = useState(false)
   629→  const [inviteEmail, setInviteEmail] = useState('')
   630→  const [inviteRole, setInviteRole] = useState('agent')
   631→  const [editingAgent, setEditingAgent] = useState<Agent | null>(null)
   632→  const [expandedId, setExpandedId] = useState<string | null>(null)
   633→  const [deleteConfirm, setDeleteConfirm] = useState(false)
   634→  const [editForm, setEditForm] = useState({ name: '', email: '', role: 'agent', status: 'online' })
   635→
   636→  // Generate stable mock stats per agent
   637→  const staffStats = useMemo(() => {
   638→    const stats: Record<string, { conversations: number; avgResponse: number; satisfaction: number; lastActive: string }> = {}
   639→    agents.forEach((agent) => {
   640→      // Simple deterministic hash from agent id
   641→      const hash = agent.id.split('').reduce((a, c) => a + c.charCodeAt(0), 0)
   642→      const conversations = 100 + (hash * 7) % 500
   643→      const avgResponse = 60 + (hash * 13) % 120 // 1-3 min in seconds
   644→      const satisfaction = 4.0 + ((hash * 3) % 10) / 10 // 4.0-5.0
   645→      const daysAgo = (hash * 11) % 30
   646→      const hoursAgo = (hash * 7) % 24
   647→      const lastActive = daysAgo === 0
   648→        ? t('common.time.hoursAgo', { h: hoursAgo })
   649→        : t('common.time.daysAgo', { d: daysAgo })
   650→      stats[agent.id] = { conversations, avgResponse, satisfaction, lastActive }
   651→    })
   652→    return stats
   653→  }, [agents])
   654→
   655→  useEffect(() => {
   656→    if (agents.length > 0) { setLoading(false); return }
   657→    async function load() {
   658→      try {
   659→        const res = await fetch('/api/agents')
   660→        const data = await res.json()
   661→        if (Array.isArray(data)) { setAgents(data) }
   662→      } catch (e) { console.error('Failed', e) }
   663→      finally { setLoading(false) }
   664→    }
   665→    load()
   666→  }, [agents.length, setAgents])
   667→
   668→  const handleInvite = () => {
   669→    if (!inviteEmail.trim()) return
   670→    const newAgent: Agent = { id: `agent_${Date.now()}`, name: inviteEmail.split('@')[0], email: inviteEmail, avatar: null, role: inviteRole, status: 'offline' }
   671→    setAgents([...agents, newAgent])
   672→    setInviteEmail('')
   673→    setShowInvite(false)
   674→  }
   675→
   676→  const openEditDialog = (agent: Agent, e?: React.MouseEvent) => {
   677→    if (e) e.stopPropagation()
   678→    setEditForm({ name: agent.name, email: agent.email, role: agent.role, status: agent.status })
   679→    setDeleteConfirm(false)
   680→    setEditingAgent(agent)
   681→  }
   682→
   683→  const handleSaveEdit = () => {
   684→    if (!editingAgent) return
   685→    const updated = { ...editingAgent, ...editForm }
   686→    const newList = agents.map(a => a.id === editingAgent.id ? updated : a)
   687→    setAgents(newList)
   688→    setEditingAgent(null)
   689→  }
   690→
   691→  const handleDelete = () => {
   692→    if (!editingAgent) return
   693→    const newList = agents.filter(a => a.id !== editingAgent.id)
   694→    setAgents(newList)
   695→    setEditingAgent(null)
   696→    setDeleteConfirm(false)
   697→  }
   698→
   699→  const toggleExpand = (id: string) => {
   700→    setExpandedId(prev => prev === id ? null : id)
   701→  }
   702→
   703→  const roleLabel = (role: string) => role === 'admin' ? t('user.role.admin') : role === 'supervisor' ? t('user.role.supervisor') : role === 'agent' ? t('user.role.agent') : role
   704→  const roleBadge = (role: string) => role === 'admin' ? 'bg-indigo-100 text-indigo-700 dark:bg-indigo-950/40 dark:text-indigo-400' : role === 'supervisor' ? 'bg-violet-100 text-violet-700 dark:bg-violet-950/40 dark:text-violet-400' : 'bg-slate-100 text-slate-600 dark:bg-slate-800/40 dark:text-slate-400'
   705→
   706→  const formatResponseTime = (seconds: number) => {
   707→    const min = Math.floor(seconds / 60)
   708→    const sec = seconds % 60
   709→    return min > 0 ? `~${min}m ${sec > 0 ? `${sec}s` : ''}` : `~${sec}s`
   710→  }
   711→
   712→  return (
   713→    <div className="space-y-6 animate-fade-in">
   714→      <div className="flex items-center justify-between">

   718→        </div>
   719→        <Button size="sm" className="h-8 rounded-xl text-xs gap-1.5 bg-gradient-to-r from-indigo-500 to-violet-500 hover:from-indigo-600 hover:to-violet-600" onClick={() => setShowInvite(!showInvite)}>

  1175→function BackupTab() {
  1176→  const { t } = useT()
  1177→  const [backups, setBackups] = useState<BackupItem[]>([])
  1178→  const [loading, setLoading] = useState(true)
  1179→  const [creating, setCreating] = useState(false)
  1180→  const [restoring, setRestoring] = useState<string | null>(null)
  1181→  const [deleting, setDeleting] = useState<string | null>(null)
  1182→  const [message, setMessage] = useState<{ type: 'success' | 'error'; text: string } | null>(null)
  1183→  const [confirmAction, setConfirmAction] = useState<{ type: 'delete' | 'restore'; id: string } | null>(null)
  1184→
  1185→  const loadBackups = async () => {
  1186→    setLoading(true)
  1187→    try {
  1188→      const res = await fetch('/api/backup')
  1189→      const data = await res.json()
  1190→      setBackups(data.data || [])
  1191→    } catch (e) { console.error(e) }
  1192→    finally { setLoading(false) }
  1193→  }
  1194→
  1195→  useEffect(() => { loadBackups() }, [])
  1196→
  1197→  const createBackup = async () => {
  1198→    setCreating(true)
  1199→    setMessage(null)
  1200→    try {
  1201→      const data: any = await apiPost('/api/backup', {})
  1202→      setMessage({ type: 'success', text: t('backup.created') })
  1203→      loadBackups()
  1204→    } catch (e: any) {
  1205→      setMessage({ type: 'error', text: e?.data?.error || t('backup.createFailed') })
  1206→    }
  1207→    finally { setCreating(false) }
  1208→  }
  1209→
  1210→  const restoreBackup = async (id: string) => {
  1211→    setRestoring(id)
  1212→    setMessage(null)
  1213→    try {
  1214→      await apiPost('/api/backup/restore', { backupId: id })
  1215→      setMessage({ type: 'success', text: t('backup.restored') })
  1216→    } catch (e: any) {
  1217→      setMessage({ type: 'error', text: e?.data?.error || t('backup.restoreFailed') })
  1218→    }
  1219→    finally { setRestoring(null) }
  1220→  }
  1221→
  1222→  const deleteBackup = async (id: string) => {
  1223→    setDeleting(id)
  1224→    try {
  1225→      await apiFetch(`/api/backup?id=${id}`, { method: 'DELETE' })
  1226→      setBackups(prev => prev.filter(b => b.id !== id))
  1227→      setMessage({ type: 'success', text: t('backup.deleteSuccess') })
  1228→    } catch {
  1229→      setMessage({ type: 'error', text: t('backup.deleteFailed') })
  1230→    }
  1231→    finally { setDeleting(null) }
  1232→  }
  1233→
  1234→  const executeConfirmedAction = async () => {
  1235→    if (!confirmAction) return
  1236→    if (confirmAction.type === 'restore') {
  1237→      setConfirmAction(null)
  1238→      await restoreBackup(confirmAction.id)
  1239→    } else {
  1240→      setConfirmAction(null)
  1241→      await deleteBackup(confirmAction.id)
  1242→    }
  1243→  }
  1244→
  1245→  const formatSize = (bytes: number) => {
  1246→    if (bytes < 1024) return `${bytes} B`
  1247→    if (bytes < 1048576) return `${(bytes / 1024).toFixed(1)} KB`
  1248→    return `${(bytes / 1048576).toFixed(1)} MB`
  1249→  }
  1250→
  1251→  if (loading) {

  1375→        </AlertDialogContent>
