   401→// ═══ STAFF TAB ═══
   402→function StaffTab() {
   403→  const { agents, setAgents } = useCRMStore()
   404→  const { t } = useT()
   405→  const [staffList, setStaffList] = useState<Agent[]>([])
   406→  const [loading, setLoading] = useState(true)
   407→  const [showInvite, setShowInvite] = useState(false)
   408→  const [inviteEmail, setInviteEmail] = useState('')
   409→  const [inviteRole, setInviteRole] = useState('agent')
   410→
   411→  useEffect(() => {
   412→    async function load() {
   413→      try {
   414→        const res = await fetch('/api/agents')
   415→        const data = await res.json()
   416→        if (Array.isArray(data)) { setStaffList(data); setAgents(data) }
   417→      } catch (e) { console.error('Failed', e) }
   418→      finally { setLoading(false) }
   419→    }
   420→    load()
   421→  }, [setAgents])
   422→
   423→  const handleInvite = () => {
   424→    if (!inviteEmail.trim()) return
   425→    const newAgent: Agent = { id: `agent_${Date.now()}`, name: inviteEmail.split('@')[0], email: inviteEmail, avatar: null, role: inviteRole, status: 'offline' }
   426→    setStaffList(prev => [...prev, newAgent])
   427→    setAgents([...staffList, newAgent])
   428→    setInviteEmail('')
   429→    setShowInvite(false)
   430→  }
   431→
   432→  const roleLabel = (role: string) => role === 'admin' ? t('user.role.admin') : role === 'supervisor' ? t('user.role.supervisor') : role === 'agent' ? t('user.role.agent') : role
   433→  const roleBadge = (role: string) => role === 'admin' ? 'bg-indigo-100 text-indigo-700 dark:bg-indigo-950/40 dark:text-indigo-400' : role === 'supervisor' ? 'bg-violet-100 text-violet-700 dark:bg-violet-950/40 dark:text-violet-400' : 'bg-slate-100 text-slate-600 dark:bg-slate-800/40 dark:text-slate-400'
   434→
   435→  return (
   436→    <div className="space-y-6 animate-fade-in">
   437→      <div className="flex items-center justify-between">
   438→        <div>
   439→          <h3 className="text-sm font-bold">{t('staff.title')}</h3>
   440→          <p className="text-xs text-muted-foreground/60 mt-1">{t('staff.count', { count: staffList.length })}</p>
   441→        </div>
   442→        <Button size="sm" className="h-8 rounded-xl text-xs gap-1.5 bg-gradient-to-r from-indigo-500 to-violet-500 hover:from-indigo-600 hover:to-violet-600" onClick={() => setShowInvite(!showInvite)}>
   443→          <Plus className="h-3.5 w-3.5" /> {t('staff.invite')}
   444→        </Button>
   445→      </div>
   446→      {showInvite && (
   447→        <div className="glass-card rounded-2xl p-5 animate-slide-down">
   448→          <h4 className="text-[13px] font-semibold mb-3">{t('staff.inviteTitle')}</h4>
   449→          <div className="flex gap-2">
   450→            <Input value={inviteEmail} onChange={(e) => setInviteEmail(e.target.value)} placeholder="email@company.com" className="flex-1 rounded-xl glass-input h-9 text-sm" onKeyDown={(e) => e.key === 'Enter' && handleInvite()} />
   451→            <Select value={inviteRole} onValueChange={setInviteRole}>
   452→              <SelectTrigger className="w-28 h-9 rounded-xl text-xs"><SelectValue /></SelectTrigger>
   453→              <SelectContent><SelectItem value="agent">{t('user.role.agent')}</SelectItem><SelectItem value="supervisor">{t('user.role.supervisor')}</SelectItem><SelectItem value="admin">{t('user.role.admin')}</SelectItem></SelectContent>
   454→            </Select>
   455→            <Button size="sm" onClick={handleInvite} className="h-9 rounded-xl text-xs"><Send className="h-3.5 w-3.5 mr-1" /> {t('staff.send')}</Button>
   456→          </div>
   457→        </div>
   458→      )}
   459→      <div className="space-y-2">
   460→        {loading ? (
   461→          <div className="space-y-3">{[1, 2, 3, 4].map((i) => (<div key={i} className="glass-card rounded-xl p-4 flex items-center gap-3"><div className="skeleton-line h-10 w-10 rounded-full flex-shrink-0" /><div className="flex-1 space-y-2"><div className="skeleton-line h-3.5 w-1/3" /><div className="skeleton-line h-3 w-1/4" /></div></div>))}</div>
   462→        ) : (
   463→          staffList.map((agent, idx) => (
   464→            <div key={agent.id} className="glass-card rounded-xl p-4 flex items-center gap-3.5 transition-all duration-200 hover:shadow-sm card-lift">
   465→              <Avatar className={cn('h-10 w-10', GRADIENT_CLASSES[idx % GRADIENT_CLASSES.length])}>
   466→                <AvatarFallback className="text-xs text-white font-semibold">{agent.name.split(' ').slice(-2).map(n => n[0]).join('')}</AvatarFallback>
   467→              </Avatar>
   468→              <div className="flex-1 min-w-0">
   469→                <div className="flex items-center gap-2">
   470→                  <span className="text-[13px] font-semibold truncate">{agent.name}</span>
   471→                  <div className={cn('h-2 w-2 rounded-full flex-shrink-0', agent.status === 'online' ? 'bg-emerald-500' : agent.status === 'busy' ? 'bg-amber-500' : 'bg-gray-400')} />
   472→                </div>
   473→                <p className="text-[11px] text-muted-foreground/50 truncate mt-0.5">{agent.email}</p>
   474→              </div>
   475→              <Badge className={cn('text-[10px] px-2 py-0.5 h-[20px] rounded-md font-medium', roleBadge(agent.role))}>{roleLabel(agent.role)}</Badge>
   476→              <Button variant="ghost" size="icon" className="h-8 w-8 rounded-lg text-muted-foreground/40 hover:text-foreground hover:bg-foreground/5"><MoreVertical className="h-4 w-4" /></Button>
   477→            </div>
   478→          ))
   479→        )}
   480→      </div>
   481→    </div>
   482→  )
