   346→// ═══ CHANNELS TAB ═══
   347→function ChannelsTab() {
   348→  const { t } = useT()
   349→  const [channels, setChannels] = useState<ChannelData[]>([])
   350→  const [loading, setLoading] = useState(true)
   351→  const [editingChannel, setEditingChannel] = useState<string | null>(null)
   352→  const [editForm, setEditForm] = useState<Record<string, string>>({})
   353→  const [saving, setSaving] = useState(false)
   354→  const [testingKey, setTestingKey] = useState<string | null>(null)
   355→  const [testResults, setTestResults] = useState<Record<string, { ok: boolean; msg: string } | null>>({})
   356→
   357→  // Load channels from API
   358→  useEffect(() => {
   359→    fetch('/api/channels')
   360→      .then(r => r.json())
   361→      .then(data => { if (Array.isArray(data)) setChannels(data) })
   362→      .catch(() => {})
   363→      .finally(() => setLoading(false))
   364→  }, [])
   365→
   366→  const toggleChannel = async (key: string) => {
   367→    const ch = channels.find(c => c.key === key)
   368→    if (!ch) return
   369→    setChannels(prev => prev.map(c => c.key === key ? { ...c, enabled: !c.enabled } : c))
   370→    apiPut('/api/channels', { channel: key, enabled: !ch.enabled, config: ch.config }).catch(() => {})
   371→  }
   372→
   373→  const startEdit = (ch: ChannelData) => {
   374→    setEditingChannel(ch.key)
   375→    setEditForm({ ...ch.config })
   376→    setTestResults(prev => ({ ...prev, [ch.key]: null }))
   377→  }
   378→
   379→  const saveChannel = async (key: string) => {
   380→    setSaving(true)
   381→    try {
   382→      const data = await apiPut('/api/channels', { channel: key, config: editForm })
   383→      setChannels(prev => prev.map(c => c.key === key ? { ...c, configured: data.configured, config: data.config || c.config } : c))
   384→      setEditingChannel(null)
   385→    } catch {}
   386→    setSaving(false)
   387→  }
   388→
   389→  const testConnection = async (key: string) => {
   390→    setTestingKey(key)
   391→    setTestResults(prev => ({ ...prev, [key]: null }))
   392→    try {
   393→      const ch = channels.find(c => c.key === key)
   394→      const result = await apiPost('/api/channels/test', { channel: key, config: { ...ch?.config, ...editForm } })
   395→      setTestResults(prev => ({ ...prev, [key]: result }))
   396→      // Update channel data to reflect new test result
   397→      setChannels(prev => prev.map(c => c.key === key ? {
   398→        ...c,
   399→        lastTestAt: new Date().toISOString(),
   400→        lastTestOk: result.ok,
   401→        lastTestMsg: result.msg,
   402→      } : c))
   403→    } catch {
   404→      setTestResults(prev => ({ ...prev, [key]: { ok: false, msg: t('common.error.serverConnection') } }))
   405→    }
   406→    setTestingKey(null)
   407→  }

   583→        : t('common.time.daysAgo', { d: daysAgo })
