   851→      {convo.status !== 'resolved' && convo.status !== 'closed' && convo.status !== 'spam' ? (
   852→        <div className={cn(
   853→          'composer-area p-3 flex-shrink-0 transition-all duration-300',
   854→          isFocused && 'shadow-[0_-8px_32px_oklch(0.49_0.2_265/0.06)]'

   856→          <div className="max-w-3xl mx-auto">

   859→              'bg-muted/50 border border-border/40',
   860→              isFocused && 'bg-background border-primary/20 shadow-lg shadow-primary/5'

   862→              <div className="flex gap-0.5 pl-1 pb-0.5">
   863→                <TooltipProvider>
   864→                  <Tooltip>
   865→                    <TooltipTrigger asChild>
   866→                      <Button variant="ghost" size="icon" className="h-8 w-8 rounded-xl text-muted-foreground/60 hover:text-muted-foreground hover:bg-foreground/5 transition-all duration-200"

   868→                        <Paperclip className="h-4 w-4" />

   870→                    </TooltipTrigger>
   871→                    <TooltipContent>{t('chat.attachFile')}</TooltipContent>
   872→                  </Tooltip>
   873→                </TooltipProvider>
   874→                <TooltipProvider>
   875→                  <Tooltip>
   876→                    <TooltipTrigger asChild>
   877→                      <Button variant="ghost" size="icon" className="h-8 w-8 rounded-xl text-muted-foreground/60 hover:text-muted-foreground hover:bg-foreground/5 transition-all duration-200"

   886→                <input ref={imageInputRef} type="file" className="hidden" accept="image/*" multiple onChange={(e) => handleFileSelect(e, 'image')} />
   887→              </div>
   888→              <Textarea
   889→                ref={textareaRef}
   890→                value={replyText}
   891→                onChange={(e) => setReplyText(e.target.value)}

   903→                      <SmilePlus className="h-4 w-4" />

   906→                  <PopoverContent side="top" align="end" className="w-[280px] p-2 rounded-xl">
   907→                    <div className="grid grid-cols-8 gap-0.5">

   913→                        >
   914→                          {emoji}
   915→                        </button>
   916→                      ))}
   917→                    </div>
   918→                  </PopoverContent>
   919→                </Popover>
   920→                <Button
   921→                  onClick={handleSend}
   922→                  disabled={(!replyText.trim() && attachedFiles.length === 0) || isSendingMessage}
   923→                  size="icon" className={cn(
   924→                    'h-9 w-9 flex-shrink-0 rounded-xl send-btn',
   925→                    (replyText.trim() || attachedFiles.length > 0) && 'bg-gradient-to-r from-indigo-500 to-violet-500 hover:from-indigo-600 hover:to-violet-600'
   926→                  )}
   927→                >
   928→                  {isSendingMessage ? (
   929→                    <div className="h-4 w-4 border-2 border-white border-t-transparent rounded-full animate-spin" />

   931→                    <Send className="h-4 w-4" />
   932→                  )}
   933→                </Button>
   934→              </div>
   935→            </div>
   936→            <p className="text-[10px] text-muted-foreground/30 text-center mt-1.5 font-medium">
   937→              {t('chat.sendHint')}
   938→            </p>
   939→          </div>
   940→        </div>
   941→      ) : (
   942→        <div className="composer-area p-4 flex-shrink-0">
   943→          <div className="max-w-3xl mx-auto flex items-center justify-between">
   944→            <span className="text-xs text-muted-foreground/60">
   945→              {convo.status === 'resolved' ? t('chat.resolved') : convo.status === 'closed' ? t('chat.closed') : t('chat.spam')}
   946→            </span>
   947→            <Button variant="outline" size="sm" className="text-xs h-8 rounded-xl font-medium hover:bg-primary hover:text-primary-foreground transition-all duration-200" onClick={() => handleStatusChange('open')}>
   948→              {t('chat.reopen')}
   949→            </Button>
   950→          </div>
   951→        </div>
   952→      )}
   953→    </div>
