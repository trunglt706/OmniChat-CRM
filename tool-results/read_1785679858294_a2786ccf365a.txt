   675→interface BlacklistItem {
   676→  type: 'ip' | 'email'
   677→  value: string
   678→  reason: string
   679→  addedAt: string
   680→  addedBy: string

   683→function SecurityTab() {
   684→  const { t } = useT()
   685→  const [config, setConfig] = useState({ rateLimitPerMinute: 60, rateLimitEnabled: true, blacklistEnabled: true })
   686→  const [blacklist, setBlacklist] = useState<BlacklistItem[]>([])
   687→  const [loading, setLoading] = useState(true)
   688→  const [newType, setNewType] = useState<'ip' | 'email'>('ip')
   689→  const [newValue, setNewValue] = useState('')
   690→  const [newReason, setNewReason] = useState('')
   691→  const [adding, setAdding] = useState(false)
   692→  const [saving, setSaving] = useState(false)
   693→  const [rateLimitInput, setRateLimitInput] = useState('60')
   694→
   695→  const loadData = async () => {
   696→    setLoading(true)
   697→    try {
   698→      const [configRes, blacklistRes] = await Promise.all([
   699→        fetch('/api/security/config'),
   700→        fetch('/api/security/blacklist'),
   701→      ])
   702→      const configData = await configRes.json()
   703→      const blacklistData = await blacklistRes.json()
   704→      setConfig(configData)
   705→      setRateLimitInput(String(configData.rateLimitPerMinute))
   706→      setBlacklist(blacklistData.data || [])
   707→    } catch (e) { console.error('Failed to load security config', e) }
   708→    finally { setLoading(false) }
   709→  }
   710→
   711→  useEffect(() => { loadData() }, [])
   712→
   713→  const saveConfig = async () => {
   714→    setSaving(true)
   715→    try {
   716→      const res = await fetch('/api/security/config', {
   717→        method: 'PUT',
   718→        headers: { 'Content-Type': 'application/json' },
   719→        body: JSON.stringify({
   720→          ...config,
   721→          rateLimitPerMinute: parseInt(rateLimitInput) || 60,
   722→        }),
   723→      })
   724→      const data = await res.json()
   725→      if (data.data) setConfig(data.data)
   726→    } catch (e) { console.error(e) }
   727→    finally { setSaving(false) }
   728→  }
   729→
   730→  const addToBlacklist = async () => {
   731→    if (!newValue.trim()) return
   732→    setAdding(true)
   733→    try {
   734→      const res = await fetch('/api/security/blacklist', {
   735→        method: 'POST',
   736→        headers: { 'Content-Type': 'application/json' },
   737→        body: JSON.stringify({ type: newType, value: newValue.trim(), reason: newReason.trim() }),
   738→      })
   739→      if (res.ok) {
   740→        const data = await res.json()
   741→        setBlacklist(prev => [data.data, ...prev])
   742→        setNewValue('')
   743→        setNewReason('')
   744→      }
   745→    } catch (e) { console.error(e) }
   746→    finally { setAdding(false) }
   747→  }
   748→

   924→function BackupTab() {
   925→  const { t } = useT()
   926→  const [backups, setBackups] = useState<BackupItem[]>([])
   927→  const [loading, setLoading] = useState(true)
   928→  const [creating, setCreating] = useState(false)
   929→  const [restoring, setRestoring] = useState<string | null>(null)
   930→  const [deleting, setDeleting] = useState<string | null>(null)
   931→  const [message, setMessage] = useState<{ type: 'success' | 'error'; text: string } | null>(null)
   932→
   933→  const loadBackups = async () => {
   934→    setLoading(true)
   935→    try {
   936→      const res = await fetch('/api/backup')
   937→      const data = await res.json()
   938→      setBackups(data.data || [])
   939→    } catch (e) { console.error(e) }
   940→    finally { setLoading(false) }
   941→  }
   942→
   943→  useEffect(() => { loadBackups() }, [])
   944→
   945→  const createBackup = async () => {
   946→    setCreating(true)
   947→    setMessage(null)
   948→    try {
   949→      const res = await fetch('/api/backup', { method: 'POST' })
   950→      const data = await res.json()
   951→      if (res.ok) {
   952→        setMessage({ type: 'success', text: t('backup.created') })
   953→        loadBackups()
   954→      } else {
   955→        setMessage({ type: 'error', text: data.error || t('backup.createFailed') })
   956→      }
   957→    } catch (e) {
   958→      setMessage({ type: 'error', text: t('backup.createFailed') })
   959→    }
   960→    finally { setCreating(false) }
   961→  }
   962→
   963→  const restoreBackup = async (id: string) => {
   964→    setRestoring(id)
   965→    setMessage(null)
   966→    try {
   967→      const res = await fetch('/api/backup/restore', {
   968→        method: 'POST',
   969→        headers: { 'Content-Type': 'application/json' },
   970→        body: JSON.stringify({ backupId: id }),
   971→      })
   972→      const data = await res.json()
   973→      if (res.ok) {
   974→        setMessage({ type: 'success', text: t('backup.restored') })
   975→      } else {
   976→        setMessage({ type: 'error', text: data.error || t('backup.restoreFailed') })
   977→      }
   978→    } catch (e) {
   979→      setMessage({ type: 'error', text: t('backup.restoreFailed') })
   980→    }
   981→    finally { setRestoring(null) }
   982→  }
   983→
   984→  const deleteBackup = async (id: string) => {
   985→    setDeleting(id)
   986→    try {
   987→      await fetch(`/api/backup?id=${id}`, { method: 'DELETE' })
   988→      setBackups(prev => prev.filter(b => b.id !== id))
   989→    } catch (e) { console.error(e) }
   990→    finally { setDeleting(null) }
   991→  }
   992→
   993→  const formatSize = (bytes: number) => {
   994→    if (bytes < 1024) return `${bytes} B`
   995→    if (bytes < 1048576) return `${(bytes / 1024).toFixed(1)} KB`
   996→    return `${(bytes / 1048576).toFixed(1)} MB`
   997→  }

  1000→    return (
  1001→      <div className="flex items-center justify-center py-20">
  1002→        <Loader2 className="h-6 w-6 animate-spin text-muted-foreground/40" />
  1003→      </div>
  1004→    )
  1005→  }
  1006→
  1007→  return (
  1008→    <div className="space-y-6 animate-fade-in">
  1009→      <div className="space-y-1">
  1010→        <SectionHeader title={t('backup.manage')} />
  1011→        <div className="glass-card rounded-2xl p-5 space-y-4">

  1014→            <HardDrive className="h-5 w-5 text-muted-foreground/40 mt-0.5 flex-shrink-0" />
  1015→            <div>
  1016→              <p className="text-xs font-medium">{t('backup.info')}</p>
  1017→              <p className="text-[10px] text-muted-foreground/40 mt-1 leading-relaxed">{t('backup.infoDesc')}</p>
  1018→            </div>
  1019→          </div>
  1020→
  1021→          {/* Create backup button */}
  1022→          <Button
  1023→            onClick={createBackup}
  1024→            disabled={creating}
  1025→            className="w-full h-10 rounded-xl text-sm font-medium bg-gradient-to-r from-indigo-500 to-violet-500 hover:from-indigo-600 hover:to-violet-600 shadow-lg shadow-indigo-500/15"
  1026→          >

  1034→              'p-3 rounded-xl text-xs font-medium flex items-center gap-2',
  1035→              message.type === 'success' ? 'bg-emerald-50 text-emerald-700 dark:bg-emerald-950/30 dark:text-emerald-400' : 'bg-red-50 text-red-700 dark:bg-red-950/30 dark:text-red-400'
