   421→    const stats: Record<string, { conversations: number; avgResponse: number; satisfaction: number; lastActive: string }> = {}
   422→    staffList.forEach((agent) => {
   423→      // Simple deterministic hash from agent id
   424→      const hash = agent.id.split('').reduce((a, c) => a + c.charCodeAt(0), 0)
   425→      const conversations = 100 + (hash * 7) % 500
   426→      const avgResponse = 60 + (hash * 13) % 120 // 1-3 min in seconds
   427→      const satisfaction = 4.0 + ((hash * 3) % 10) / 10 // 4.0-5.0
   428→      const daysAgo = (hash * 11) % 30
   429→      const hoursAgo = (hash * 7) % 24
   430→      const lastActive = daysAgo === 0
   431→        ? `${hoursAgo}h ago`
   432→        : `${daysAgo}d ago`
   433→      stats[agent.id] = { conversations, avgResponse, satisfaction, lastActive }
   434→    })

   439→    async function load() {
   440→      try {
   441→        const res = await fetch('/api/agents')

   489→  const roleBadge = (role: string) => role === 'admin' ? 'bg-indigo-100 text-indigo-700 dark:bg-indigo-950/40 dark:text-indigo-400' : role === 'supervisor' ? 'bg-violet-100 text-violet-700 dark:bg-violet-950/40 dark:text-violet-400' : 'bg-slate-100 text-slate-600 dark:bg-slate-800/40 dark:text-slate-400'
   490→

   495→  }

   524→        ) : (
