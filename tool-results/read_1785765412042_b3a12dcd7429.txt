    53→function shouldShowDate(messages: Message[], idx: number) {
    54→  if (idx === 0) return true
    55→  const prev = new Date(messages[idx - 1].createdAt).toDateString()
    56→  const curr = new Date(messages[idx].createdAt).toDateString()
    57→  return prev !== curr
    58→}

   317→      }
   318→    } catch (e) {
   319→      console.error('Failed to load older messages', e)
