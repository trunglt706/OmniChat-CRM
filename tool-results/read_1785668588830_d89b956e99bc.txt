    51→function shouldShowDate(messages: Message[], idx: number) {
    52→  if (idx === 0) return true
    53→  const prev = new Date(messages[idx - 1].createdAt).toDateString()
    54→  const curr = new Date(messages[idx].createdAt).toDateString()
    55→  return prev !== curr
    56→}
