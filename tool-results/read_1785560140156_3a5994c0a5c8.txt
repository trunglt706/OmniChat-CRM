    50→function shouldShowDate(messages: Message[], idx: number) {
    51→  if (idx === 0) return true
    52→  const prev = new Date(messages[idx - 1].createdAt).toDateString()
    53→  const curr = new Date(messages[idx].createdAt).toDateString()
    54→  return prev !== curr
    55→}

   229→      return null

   231→  }, [])
   232→

   234→  useEffect(() => {
   235→    if (!selectedConversationId) return
   236→    isInitialLoadRef.current = true
   237→    prevMessageCountRef.current = 0
   238→

   240→      try {

   251→        setMessages(msgsJson.data || [])

   254→

   259→        })
   260→      } catch (e) {
   261→        console.error('Failed to fetch conversation', e)
   262→      }

   266→

   276→  // ─── Load older messages on scroll up ───

   295→        setHasMoreMessages(result.hasMore || false)

   299→          if (container) {
   300→            const newScrollHeight = container.scrollHeight
   301→            container.scrollTop = newScrollHeight - prevScrollHeight + prevScrollTop
   302→          }
   303→        })
   304→      } else {
   305→        setHasMoreMessages(false)
   306→      }
   307→    } catch (e) {
   308→      console.error('Failed to load older messages', e)
   309→    } finally {
   310→      loadingMoreRef.current = false
   311→      setIsLoadingMoreMessages(false)
   312→    }
   313→  }, [selectedConversationId, hasMoreMessages, isLoadingMoreMessages, messages, fetchMessages, prependMessages, setHasMoreMessages, setIsLoadingMoreMessages])
