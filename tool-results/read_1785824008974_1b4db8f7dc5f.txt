   344→// ═══ CHANNELS TAB ═══
   345→function ChannelsTab() {
   346→  const { t } = useT()
   347→  const [channels, setChannels] = useState<ChannelData[]>([])
   348→  const [loading, setLoading] = useState(true)
   349→  const [editingChannel, setEditingChannel] = useState<string | null>(null)
   350→  const [editForm, setEditForm] = useState<Record<string, string>>({})
   351→  const [saving, setSaving] = useState(false)
   352→  const [testingKey, setTestingKey] = useState<string | null>(null)
   353→  const [testResults, setTestResults] = useState<Record<string, { ok: boolean; msg: string } | null>>({})
   354→
   355→  // Load channels from API
   356→  useEffect(() => {
   357→    fetch('/api/channels')
   358→      .then(r => r.json())
   359→      .then(data => { if (Array.isArray(data)) setChannels(data) })
   360→      .catch(() => {})
   361→      .finally(() => setLoading(false))
   362→  }, [])
   363→
   364→  const toggleChannel = async (key: string) => {
   365→    const ch = channels.find(c => c.key === key)
   366→    if (!ch) return
   367→    setChannels(prev => prev.map(c => c.key === key ? { ...c, enabled: !c.enabled } : c))
   368→    apiPut('/api/channels', { channel: key, enabled: !ch.enabled, config: ch.config }).catch(() => {})
   369→  }
   370→
   371→  const startEdit = (ch: ChannelData) => {
   372→    setEditingChannel(ch.key)
   373→    setEditForm({ ...ch.config })
   374→    setTestResults(prev => ({ ...prev, [ch.key]: null }))
   375→  }
   376→
   377→  const saveChannel = async (key: string) => {
   378→    setSaving(true)
   379→    try {
   380→      const data = await apiPut('/api/channels', { channel: key, config: editForm })
   381→      setChannels(prev => prev.map(c => c.key === key ? { ...c, configured: data.configured, config: data.config || c.config } : c))
   382→      setEditingChannel(null)
   383→    } catch {}
   384→    setSaving(false)
   385→  }
   386→
   387→  const testConnection = async (key: string) => {
   388→    setTestingKey(key)
   389→    setTestResults(prev => ({ ...prev, [key]: null }))
   390→    try {
   391→      const ch = channels.find(c => c.key === key)
   392→      const result = await apiPost('/api/channels/test', { channel: key, config: { ...ch?.config, ...editForm } })
   393→      setTestResults(prev => ({ ...prev, [key]: result }))
   394→      // Update channel data to reflect new test result
   395→      setChannels(prev => prev.map(c => c.key === key ? {
   396→        ...c,
   397→        lastTestAt: new Date().toISOString(),
   398→        lastTestOk: result.ok,
   399→        lastTestMsg: result.msg,
   400→      } : c))
   401→    } catch {
   402→      setTestResults(prev => ({ ...prev, [key]: { ok: false, msg: t('common.error.serverConnection') } }))
   403→    }
   404→    setTestingKey(null)
   405→  }
   406→
   407→  if (loading) {
   408→    return (
   409→      <div className="flex items-center justify-center py-20">

   574→      const conversations = 100 + (hash * 7) % 500

   576→      const satisfaction = 4.0 + ((hash * 3) % 10) / 10 // 4.0-5.0

   578→      const hoursAgo = (hash * 7) % 24
   579→      const lastActive = daysAgo === 0
   580→        ? t('common.time.hoursAgo', { h: hoursAgo })
   581→        : t('common.time.daysAgo', { d: daysAgo })
