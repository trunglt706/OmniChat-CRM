   140→function ProfileTab() {
   141→  const currentUser = useCRMStore((s) => s.currentUser)
   142→  const setCurrentUser = useCRMStore((s) => s.setCurrentUser)
   143→  const { t } = useT()
   144→  const [editing, setEditing] = useState(false)
   145→  const [saving, setSaving] = useState(false)
   146→  const savingRef = useRef(false)
   147→  const [form, setForm] = useState({ name: '', email: '', phone: '', bio: '' })
   148→  const fileRef = useRef<HTMLInputElement>(null)
   149→  const [stats, setStats] = useState<AgentStats | null>(null)
   150→  const [statusModal, setStatusModal] = useState<{ type: 'success' | 'error'; text: string } | null>(null)
   151→
   152→  useEffect(() => {
   153→    fetch('/api/agents/me/stats')
   154→      .then(res => res.ok ? res.json() : null)
   155→      .then(data => { if (data) setStats(data) })
   156→      .catch(() => {})
   157→  }, [])
   158→
   159→  useEffect(() => {
   160→    if (currentUser) setForm({ name: currentUser.name, email: currentUser.email, phone: currentUser.phone, bio: currentUser.bio })
   161→  }, [currentUser])
   162→
   163→  const handleSave = async () => {
   164→    if (!currentUser || savingRef.current) return
   165→    savingRef.current = true
   166→    setSaving(true)
   167→    try {
   168→      const updated = await apiPut('/api/auth/me', { name: form.name, email: form.email, phone: form.phone, bio: form.bio })
   169→      setCurrentUser({ ...currentUser, ...updated })
   170→      setEditing(false)
   171→      setStatusModal({ type: 'success', text: t('profile.saveSuccess') })
   172→    } catch {
   173→      setStatusModal({ type: 'error', text: t('profile.saveFailed') })
   174→    }
   175→    setSaving(false)
   176→    savingRef.current = false
   177→  }
   178→  const handleCancel = () => { if (currentUser) setForm({ name: currentUser.name, email: currentUser.email, phone: currentUser.phone, bio: currentUser.bio }); setEditing(false) }
   179→  const handleStatusChange = async (status: UserProfile['status']) => {
   180→    if (!currentUser) return
   181→    setCurrentUser({ ...currentUser, status })
   182→    try { await apiPut('/api/auth/me', { status }) } catch { /* optimistic update already applied */ }
   183→  }
   184→
   185→  if (!currentUser) return null
   186→
   187→  return (
   188→    <div className="space-y-6 animate-fade-in">
   189→      {/* Status Modal */}
   190→      <Dialog open={!!statusModal} onOpenChange={(open) => { if (!open) setStatusModal(null) }}>
   191→        <DialogContent className="sm:max-w-sm rounded-2xl">
   192→          <DialogHeader>
   193→            <DialogTitle className={cn('text-base flex items-center gap-2', statusModal?.type === 'success' ? 'text-emerald-600' : 'text-red-600')}>
   194→              {statusModal?.type === 'success' ? <Check className="h-5 w-5" /> : <X className="h-5 w-5" />}
   195→              {statusModal?.type === 'success' ? t('common.success') : t('common.failed')}
   196→            </DialogTitle>
   197→            <DialogDescription className="text-sm text-muted-foreground/70">
   198→              {statusModal?.text}
   199→            </DialogDescription>
   200→          </DialogHeader>
   201→          <DialogFooter>
   202→            <Button onClick={() => setStatusModal(null)} className="rounded-xl h-9 text-xs">{t('common.close')}</Button>
   203→          </DialogFooter>
   204→        </DialogContent>
   205→      </Dialog>
   206→      <div className="glass-card rounded-2xl p-6">
   207→        <div className="flex flex-col items-center gap-4">
   208→          <div className="relative group">
   209→            <Avatar className={cn('h-24 w-24 ring-4 ring-background shadow-xl', GRADIENT_CLASSES[0])}>
   210→              <AvatarFallback className="text-3xl text-white font-bold">{currentUser.name.split(' ').slice(-2).map(n => n[0]).join('')}</AvatarFallback>
   211→            </Avatar>
   212→            <button onClick={() => fileRef.current?.click()} className="absolute inset-0 rounded-full bg-black/0 group-hover:bg-black/30 transition-all duration-200 flex items-center justify-center opacity-0 group-hover:opacity-100">
   213→              <Camera className="h-6 w-6 text-white" />
   214→            </button>
   215→            <input ref={fileRef} type="file" accept="image/*" className="hidden" onChange={() => {}} />
   216→          </div>
   217→          {!editing && (
   218→            <div className="text-center">
   219→              <h3 className="text-xl font-bold tracking-tight">{currentUser.name}</h3>
   220→              <p className="text-sm text-muted-foreground/60 mt-1 font-medium">{currentUser.email}</p>
   221→              <span className="inline-flex items-center gap-1.5 mt-2.5 text-xs font-medium px-3 py-1.5 rounded-full bg-foreground/[0.04] text-muted-foreground/70">
   222→                <Shield className="h-3.5 w-3.5" />
   223→                {currentUser.role === 'admin' ? t('user.role.admin') : currentUser.role === 'agent' ? t('user.role.agent') : currentUser.role === 'supervisor' ? t('user.role.supervisor') : currentUser.role}
   224→              </span>
   225→            </div>
   226→          )}
   227→          <div className="flex gap-2">
   228→            {STATUS_OPTIONS.map((opt) => (
   229→              <button key={opt.value} onClick={() => handleStatusChange(opt.value)} className={cn('flex items-center gap-1.5 px-3.5 py-2 rounded-xl text-xs font-medium transition-all duration-200 border', currentUser.status === opt.value ? 'border-primary/30 bg-primary/5 text-primary' : 'border-transparent text-muted-foreground/50 hover:text-foreground hover:bg-foreground/[0.03]')}>
   230→                <span className={cn('h-2 w-2 rounded-full', opt.color, currentUser.status === opt.value && 'shadow-sm')} />
   231→                {t(opt.labelKey)}
   232→              </button>
   233→            ))}
   234→          </div>
   235→        </div>
   236→      </div>
   237→      <div className="space-y-4">
   238→        <SectionHeader title={t('profile.personalInfo')} />
   239→        <div className="glass-card rounded-2xl p-5 space-y-4">
   240→          <div className="space-y-3">
   241→            <div className="space-y-1.5">
   242→              <Label className="text-xs font-medium text-muted-foreground/70">{t('profile.displayName')}</Label>
   243→              <Input value={form.name} onChange={(e) => setForm(f => ({ ...f, name: e.target.value }))} disabled={!editing} className="rounded-xl glass-input h-10 text-sm" />
   244→            </div>
   245→            <div className="space-y-1.5">
   246→              <Label className="text-xs font-medium text-muted-foreground/70">{t('profile.email')}</Label>
   247→              <Input type="email" value={form.email} onChange={(e) => setForm(f => ({ ...f, email: e.target.value }))} disabled={!editing} className="rounded-xl glass-input h-10 text-sm" />
   248→            </div>
   249→            <div className="space-y-1.5">
   250→              <Label className="text-xs font-medium text-muted-foreground/70">{t('profile.phoneNumber')}</Label>
   251→              <Input value={form.phone} onChange={(e) => setForm(f => ({ ...f, phone: e.target.value }))} disabled={!editing} className="rounded-xl glass-input h-10 text-sm" />
   252→            </div>
   253→            <div className="space-y-1.5">
   254→              <Label className="text-xs font-medium text-muted-foreground/70">{t('profile.bio')}</Label>
   255→              <Textarea value={form.bio} onChange={(e) => setForm(f => ({ ...f, bio: e.target.value }))} disabled={!editing} className="rounded-xl glass-input min-h-[80px] resize-none text-sm" />
   256→            </div>
   257→          </div>
   258→          {!editing ? (
   259→            <Button onClick={() => setEditing(true)} className="w-full h-10 rounded-xl text-sm font-medium"><Pencil className="h-4 w-4 mr-1.5" /> {t('profile.editInfo')}</Button>
   260→          ) : (
   261→            <div className="flex gap-2">
   262→              <Button variant="outline" onClick={handleCancel} className="flex-1 h-10 rounded-xl text-sm font-medium"><X className="h-4 w-4 mr-1.5" /> {t('profile.cancel')}</Button>
   263→              <Button onClick={handleSave} disabled={saving} className="flex-1 h-10 rounded-xl text-sm font-medium bg-gradient-to-r from-indigo-500 to-violet-500 hover:from-indigo-600 hover:to-violet-600">{saving ? <Loader2 className="h-4 w-4 mr-1.5 animate-spin" /> : <Check className="h-4 w-4 mr-1.5" />} {t('profile.save')}</Button>
   264→            </div>
   265→          )}
   266→        </div>
   267→      </div>
   268→      <div className="space-y-2">
   269→        <SectionHeader title={t('profile.stats.title')} />
   270→        <div className="grid grid-cols-2 gap-3">
   271→          {[
   272→            { label: t('profile.stat.conversationsToday'), value: stats ? String(stats.conversationsToday) : '--', icon: MessageSquare },
   273→            { label: t('profile.stat.avgResponse'), value: stats?.avgResponse || '--', icon: Clock },
   274→            { label: t('profile.stat.avgRating'), value: stats?.avgRating || '--', icon: Shield },
   275→            { label: t('profile.stat.totalConversations'), value: stats ? String(stats.totalConversations) : '--', icon: User },
   276→          ].map((stat) => (
   277→            <div key={stat.label} className="glass-card rounded-xl p-4">
   278→              <stat.icon className="h-4 w-4 text-muted-foreground/40 mb-2" />
   279→              <p className="text-lg font-bold tabular-nums">{stat.value}</p>
   280→              <p className="text-[11px] text-muted-foreground/50 mt-0.5 font-medium">{stat.label}</p>
   281→            </div>
   282→          ))}
   283→        </div>
   284→      </div>
   285→    </div>
   286→  )
   287→}

   290→function SystemTab() {
   291→  const settings = useCRMStore((s) => s.settings)
   292→  const updateSettings = useCRMStore((s) => s.updateSettings)
   293→  const notifCount = useCRMStore((s) => s.notifications.length)
   294→  const clearAllNotifications = useCRMStore((s) => s.clearAllNotifications)
   295→  const { theme, setTheme } = useTheme()
   296→  const [mounted, setMounted] = useState(false)
   297→  const [statusModal, setStatusModal] = useState<{ type: 'success' | 'error'; text: string } | null>(null)
   298→  const { t } = useT()
   299→  useEffect(() => { setMounted(true) }, [])
   300→
   301→  const handleUpdateSetting = async (patch: Record<string, unknown>) => {
   302→    updateSettings(patch)
   303→    try {
   304→      const newSettings = { ...useCRMStore.getState().settings, ...patch }
   305→      await apiPut('/api/auth/me/settings', newSettings)
   306→      setStatusModal({ type: 'success', text: t('settings.saveSuccess') })
   307→    } catch {
   308→      setStatusModal({ type: 'error', text: t('settings.saveFailed') })
   309→    }
   310→  }
   311→
   312→  return (
   313→    <div className="space-y-6 animate-fade-in">
   314→      {/* Status Modal */}
   315→      <Dialog open={!!statusModal} onOpenChange={(open) => { if (!open) setStatusModal(null) }}>
   316→        <DialogContent className="sm:max-w-sm rounded-2xl">
   317→          <DialogHeader>
   318→            <DialogTitle className={cn('text-base flex items-center gap-2', statusModal?.type === 'success' ? 'text-emerald-600' : 'text-red-600')}>
   319→              {statusModal?.type === 'success' ? <Check className="h-5 w-5" /> : <X className="h-5 w-5" />}
   320→              {statusModal?.type === 'success' ? t('common.success') : t('common.failed')}
   321→            </DialogTitle>
   322→            <DialogDescription className="text-sm text-muted-foreground/70">{statusModal?.text}</DialogDescription>
   323→          </DialogHeader>
   324→          <DialogFooter>
   325→            <Button onClick={() => setStatusModal(null)} className="rounded-xl h-9 text-xs">{t('common.close')}</Button>
   326→          </DialogFooter>
   327→        </DialogContent>
   328→      </Dialog>
   329→      <div className="space-y-1">
   330→        <SectionHeader title={t('settings.appearance')} />
   331→        <div className="glass-card rounded-2xl p-5">
   332→          <SettingRow icon={mounted && theme === 'dark' ? Moon : Sun} label={t('settings.theme')} description={t('settings.themeDesc')}>
   333→            <Button variant="outline" size="sm" className="h-8 rounded-lg text-xs gap-1.5 border-border/40" onClick={() => setTheme(theme === 'dark' ? 'light' : 'dark')}>
   334→              {mounted && (theme === 'dark' ? <Moon className="h-3.5 w-3.5" /> : <Sun className="h-3.5 w-3.5" />)}

   341→          </SettingRow>
   342→          <Separator className="opacity-30 my-1" />
   343→          <SettingRow icon={Eye} label={t('settings.showPreview')} description={t('settings.showPreviewDesc')}>

   612→function StaffTab() {
   613→  const agents = useCRMStore((s) => s.agents)
   614→  const setAgents = useCRMStore((s) => s.setAgents)
   615→  const { t } = useT()
   616→  const [loading, setLoading] = useState(true)
   617→  const [showInvite, setShowInvite] = useState(false)
   618→  const [inviteEmail, setInviteEmail] = useState('')
   619→  const [inviteRole, setInviteRole] = useState('agent')
   620→  const [editingAgent, setEditingAgent] = useState<Agent | null>(null)
   621→  const [expandedId, setExpandedId] = useState<string | null>(null)
   622→  const [deleteConfirm, setDeleteConfirm] = useState(false)
   623→  const [editForm, setEditForm] = useState({ name: '', email: '', role: 'agent', status: 'online' })
   624→
   625→  // Generate stable mock stats per agent
   626→  const staffStats = useMemo(() => {
   627→    const stats: Record<string, { conversations: number; avgResponse: number; satisfaction: number; lastActive: string }> = {}
   628→    agents.forEach((agent) => {
   629→      // Simple deterministic hash from agent id
   630→      const hash = agent.id.split('').reduce((a, c) => a + c.charCodeAt(0), 0)
   631→      const conversations = 100 + (hash * 7) % 500
   632→      const avgResponse = 60 + (hash * 13) % 120 // 1-3 min in seconds
   633→      const satisfaction = 4.0 + ((hash * 3) % 10) / 10 // 4.0-5.0
   634→      const daysAgo = (hash * 11) % 30
   635→      const hoursAgo = (hash * 7) % 24
   636→      const lastActive = daysAgo === 0
   637→        ? t('common.time.hoursAgo', { h: hoursAgo })
   638→        : t('common.time.daysAgo', { d: daysAgo })
   639→      stats[agent.id] = { conversations, avgResponse, satisfaction, lastActive }
   640→    })
   641→    return stats
   642→  }, [agents])
   643→
   644→  useEffect(() => {
   645→    if (agents.length > 0) { setLoading(false); return }
   646→    async function load() {
   647→      try {
   648→        const res = await fetch('/api/agents')
   649→        const data = await res.json()
   650→        if (Array.isArray(data)) { setAgents(data) }
   651→      } catch (e) { console.error('Failed', e) }
   652→      finally { setLoading(false) }
   653→    }
   654→    load()
   655→  }, [agents.length, setAgents])
   656→

   658→    if (!inviteEmail.trim()) return

   660→    setAgents([...agents, newAgent])
   661→    setInviteEmail('')
   662→    setShowInvite(false)
   663→  }
   664→
   665→  const openEditDialog = (agent: Agent, e?: React.MouseEvent) => {
   666→    if (e) e.stopPropagation()
   667→    setEditForm({ name: agent.name, email: agent.email, role: agent.role, status: agent.status })
   668→    setDeleteConfirm(false)
   669→    setEditingAgent(agent)
   670→  }
   671→
   672→  const handleSaveEdit = () => {
   673→    if (!editingAgent) return
   674→    const updated = { ...editingAgent, ...editForm }
   675→    const newList = agents.map(a => a.id === editingAgent.id ? updated : a)
   676→    setAgents(newList)
   677→    setEditingAgent(null)
   678→  }
   679→
   680→  const handleDelete = () => {
   681→    if (!editingAgent) return
   682→    const newList = agents.filter(a => a.id !== editingAgent.id)
   683→    setAgents(newList)
   684→    setEditingAgent(null)
   685→    setDeleteConfirm(false)
   686→  }
   687→

   693→  const roleBadge = (role: string) => role === 'admin' ? 'bg-indigo-100 text-indigo-700 dark:bg-indigo-950/40 dark:text-indigo-400' : role === 'supervisor' ? 'bg-violet-100 text-violet-700 dark:bg-violet-950/40 dark:text-violet-400' : 'bg-slate-100 text-slate-600 dark:bg-slate-800/40 dark:text-slate-400'

   697→    const sec = seconds % 60

  1164→function BackupTab() {

  1167→  const [loading, setLoading] = useState(true)

  1170→  const [deleting, setDeleting] = useState<string | null>(null)
  1171→  const [message, setMessage] = useState<{ type: 'success' | 'error'; text: string } | null>(null)
  1172→  const [confirmAction, setConfirmAction] = useState<{ type: 'delete' | 'restore'; id: string } | null>(null)
  1173→
  1174→  const loadBackups = async () => {
  1175→    setLoading(true)
  1176→    try {
  1177→      const res = await fetch('/api/backup')
  1178→      const data = await res.json()
  1179→      setBackups(data.data || [])
  1180→    } catch (e) { console.error(e) }
  1181→    finally { setLoading(false) }
  1182→  }
  1183→
  1184→  useEffect(() => { loadBackups() }, [])
  1185→
  1186→  const createBackup = async () => {
  1187→    setCreating(true)
  1188→    setMessage(null)
  1189→    try {
  1190→      const data: any = await apiPost('/api/backup', {})
  1191→      setMessage({ type: 'success', text: t('backup.created') })
  1192→      loadBackups()
  1193→    } catch (e: any) {
  1194→      setMessage({ type: 'error', text: e?.data?.error || t('backup.createFailed') })
  1195→    }
  1196→    finally { setCreating(false) }
  1197→  }
  1198→
  1199→  const restoreBackup = async (id: string) => {
  1200→    setRestoring(id)
  1201→    setMessage(null)
  1202→    try {
  1203→      await apiPost('/api/backup/restore', { backupId: id })
  1204→      setMessage({ type: 'success', text: t('backup.restored') })
  1205→    } catch (e: any) {
  1206→      setMessage({ type: 'error', text: e?.data?.error || t('backup.restoreFailed') })
  1207→    }
  1208→    finally { setRestoring(null) }
  1209→  }
  1210→
  1211→  const deleteBackup = async (id: string) => {
  1212→    setDeleting(id)
  1213→    try {
  1214→      await apiFetch(`/api/backup?id=${id}`, { method: 'DELETE' })
  1215→      setBackups(prev => prev.filter(b => b.id !== id))
  1216→      setMessage({ type: 'success', text: t('backup.deleteSuccess') })
  1217→    } catch {
  1218→      setMessage({ type: 'error', text: t('backup.deleteFailed') })
  1219→    }
  1220→    finally { setDeleting(null) }
  1221→  }
  1222→
  1223→  const executeConfirmedAction = async () => {
  1224→    if (!confirmAction) return
  1225→    if (confirmAction.type === 'restore') {
  1226→      setConfirmAction(null)
  1227→      await restoreBackup(confirmAction.id)
  1228→    } else {
  1229→      setConfirmAction(null)
  1230→      await deleteBackup(confirmAction.id)
  1231→    }
  1232→  }
  1233→
  1234→  const formatSize = (bytes: number) => {
  1235→    if (bytes < 1024) return `${bytes} B`
  1236→    if (bytes < 1048576) return `${(bytes / 1024).toFixed(1)} KB`
  1237→    return `${(bytes / 1048576).toFixed(1)} MB`
  1238→  }
  1239→
  1240→  if (loading) {
  1241→    return (
  1242→      <div className="flex items-center justify-center py-20">
  1243→        <Loader2 className="h-6 w-6 animate-spin text-muted-foreground/40" />
  1244→      </div>
  1245→    )
  1246→  }
  1247→
  1248→  return (
  1249→    <div className="space-y-6 animate-fade-in">
  1250→      <div className="space-y-1">
  1251→        <SectionHeader title={t('backup.manage')} />
  1252→        <div className="glass-card rounded-2xl p-5 space-y-4">

  1364→        </AlertDialogContent>

  1366→    </div>
