   140→function ProfileTab() {
   141→  const currentUser = useCRMStore((s) => s.currentUser)
   142→  const setCurrentUser = useCRMStore((s) => s.setCurrentUser)
   143→  const { t } = useT()
   144→  const [editing, setEditing] = useState(false)
   145→  const [saving, setSaving] = useState(false)
   146→  const [form, setForm] = useState({ name: '', email: '', phone: '', bio: '' })
   147→  const fileRef = useRef<HTMLInputElement>(null)
   148→  const [stats, setStats] = useState<AgentStats | null>(null)
   149→  const [statusModal, setStatusModal] = useState<{ type: 'success' | 'error'; text: string } | null>(null)
   150→
   151→  useEffect(() => {
   152→    fetch('/api/agents/me/stats')
   153→      .then(res => res.ok ? res.json() : null)
   154→      .then(data => { if (data) setStats(data) })
   155→      .catch(() => {})
   156→  }, [])
   157→
   158→  useEffect(() => {
   159→    if (currentUser) setForm({ name: currentUser.name, email: currentUser.email, phone: currentUser.phone, bio: currentUser.bio })
   160→  }, [currentUser])
   161→
   162→  const handleSave = async () => {
   163→    if (!currentUser) return
   164→    setSaving(true)
   165→    try {
   166→      const updated = await apiPut('/api/auth/me', { name: form.name, email: form.email, phone: form.phone, bio: form.bio })
   167→      setCurrentUser({ ...currentUser, ...updated })
   168→      setEditing(false)
   169→      setStatusModal({ type: 'success', text: t('profile.saveSuccess') })
   170→    } catch {
   171→      setStatusModal({ type: 'error', text: t('profile.saveFailed') })
   172→    }
   173→    setSaving(false)
   174→  }
   175→  const handleCancel = () => { if (currentUser) setForm({ name: currentUser.name, email: currentUser.email, phone: currentUser.phone, bio: currentUser.bio }); setEditing(false) }
   176→  const handleStatusChange = async (status: UserProfile['status']) => {
   177→    if (!currentUser) return
   178→    setCurrentUser({ ...currentUser, status })
   179→    try { await apiPut('/api/auth/me', { status }) } catch { /* optimistic update already applied */ }
   180→  }
   181→
   182→  if (!currentUser) return null
   183→
   184→  return (
   185→    <div className="space-y-6 animate-fade-in">
   186→      {/* Status Modal */}
   187→      <Dialog open={!!statusModal} onOpenChange={(open) => { if (!open) setStatusModal(null) }}>
   188→        <DialogContent className="sm:max-w-sm rounded-2xl">
   189→          <DialogHeader>
   190→            <DialogTitle className={cn('text-base flex items-center gap-2', statusModal?.type === 'success' ? 'text-emerald-600' : 'text-red-600')}>
   191→              {statusModal?.type === 'success' ? <Check className="h-5 w-5" /> : <X className="h-5 w-5" />}
   192→              {statusModal?.type === 'success' ? t('common.success') : t('common.failed')}
   193→            </DialogTitle>
   194→            <DialogDescription className="text-sm text-muted-foreground/70">
   195→              {statusModal?.text}
   196→            </DialogDescription>
   197→          </DialogHeader>
   198→          <DialogFooter>
   199→            <Button onClick={() => setStatusModal(null)} className="rounded-xl h-9 text-xs">{t('common.close')}</Button>
   200→          </DialogFooter>
   201→        </DialogContent>
   202→      </Dialog>
   203→      <div className="glass-card rounded-2xl p-6">
   204→        <div className="flex flex-col items-center gap-4">
   205→          <div className="relative group">
   206→            <Avatar className={cn('h-24 w-24 ring-4 ring-background shadow-xl', GRADIENT_CLASSES[0])}>
   207→              <AvatarFallback className="text-3xl text-white font-bold">{currentUser.name.split(' ').slice(-2).map(n => n[0]).join('')}</AvatarFallback>
   208→            </Avatar>
   209→            <button onClick={() => fileRef.current?.click()} className="absolute inset-0 rounded-full bg-black/0 group-hover:bg-black/30 transition-all duration-200 flex items-center justify-center opacity-0 group-hover:opacity-100">
   210→              <Camera className="h-6 w-6 text-white" />
   211→            </button>
   212→            <input ref={fileRef} type="file" accept="image/*" className="hidden" onChange={() => {}} />
   213→          </div>
   214→          {!editing && (
   215→            <div className="text-center">
   216→              <h3 className="text-xl font-bold tracking-tight">{currentUser.name}</h3>
   217→              <p className="text-sm text-muted-foreground/60 mt-1 font-medium">{currentUser.email}</p>
   218→              <span className="inline-flex items-center gap-1.5 mt-2.5 text-xs font-medium px-3 py-1.5 rounded-full bg-foreground/[0.04] text-muted-foreground/70">
   219→                <Shield className="h-3.5 w-3.5" />
   220→                {currentUser.role === 'admin' ? t('user.role.admin') : currentUser.role === 'agent' ? t('user.role.agent') : currentUser.role === 'supervisor' ? t('user.role.supervisor') : currentUser.role}
   221→              </span>
   222→            </div>
   223→          )}
   224→          <div className="flex gap-2">
   225→            {STATUS_OPTIONS.map((opt) => (
   226→              <button key={opt.value} onClick={() => handleStatusChange(opt.value)} className={cn('flex items-center gap-1.5 px-3.5 py-2 rounded-xl text-xs font-medium transition-all duration-200 border', currentUser.status === opt.value ? 'border-primary/30 bg-primary/5 text-primary' : 'border-transparent text-muted-foreground/50 hover:text-foreground hover:bg-foreground/[0.03]')}>
   227→                <span className={cn('h-2 w-2 rounded-full', opt.color, currentUser.status === opt.value && 'shadow-sm')} />
   228→                {t(opt.labelKey)}
   229→              </button>
   230→            ))}
   231→          </div>
   232→        </div>
   233→      </div>
   234→      <div className="space-y-4">
   235→        <SectionHeader title={t('profile.personalInfo')} />
   236→        <div className="glass-card rounded-2xl p-5 space-y-4">
   237→          <div className="space-y-3">
   238→            <div className="space-y-1.5">
   239→              <Label className="text-xs font-medium text-muted-foreground/70">{t('profile.displayName')}</Label>
   240→              <Input value={form.name} onChange={(e) => setForm(f => ({ ...f, name: e.target.value }))} disabled={!editing} className="rounded-xl glass-input h-10 text-sm" />
   241→            </div>
   242→            <div className="space-y-1.5">
   243→              <Label className="text-xs font-medium text-muted-foreground/70">{t('profile.email')}</Label>
   244→              <Input type="email" value={form.email} onChange={(e) => setForm(f => ({ ...f, email: e.target.value }))} disabled={!editing} className="rounded-xl glass-input h-10 text-sm" />
   245→            </div>
   246→            <div className="space-y-1.5">
   247→              <Label className="text-xs font-medium text-muted-foreground/70">{t('profile.phoneNumber')}</Label>
   248→              <Input value={form.phone} onChange={(e) => setForm(f => ({ ...f, phone: e.target.value }))} disabled={!editing} className="rounded-xl glass-input h-10 text-sm" />
   249→            </div>
   250→            <div className="space-y-1.5">
   251→              <Label className="text-xs font-medium text-muted-foreground/70">{t('profile.bio')}</Label>
   252→              <Textarea value={form.bio} onChange={(e) => setForm(f => ({ ...f, bio: e.target.value }))} disabled={!editing} className="rounded-xl glass-input min-h-[80px] resize-none text-sm" />
   253→            </div>
   254→          </div>
   255→          {!editing ? (
   256→            <Button onClick={() => setEditing(true)} className="w-full h-10 rounded-xl text-sm font-medium"><Pencil className="h-4 w-4 mr-1.5" /> {t('profile.editInfo')}</Button>
   257→          ) : (
   258→            <div className="flex gap-2">
   259→              <Button variant="outline" onClick={handleCancel} className="flex-1 h-10 rounded-xl text-sm font-medium"><X className="h-4 w-4 mr-1.5" /> {t('profile.cancel')}</Button>
   260→              <Button onClick={handleSave} disabled={saving} className="flex-1 h-10 rounded-xl text-sm font-medium bg-gradient-to-r from-indigo-500 to-violet-500 hover:from-indigo-600 hover:to-violet-600">{saving ? <Loader2 className="h-4 w-4 mr-1.5 animate-spin" /> : <Check className="h-4 w-4 mr-1.5" />} {t('profile.save')}</Button>
   261→            </div>
   262→          )}
   263→        </div>
   264→      </div>
   265→      <div className="space-y-2">
   266→        <SectionHeader title={t('profile.stats.title')} />
   267→        <div className="grid grid-cols-2 gap-3">
   268→          {[
   269→            { label: t('profile.stat.conversationsToday'), value: stats ? String(stats.conversationsToday) : '--', icon: MessageSquare },
   270→            { label: t('profile.stat.avgResponse'), value: stats?.avgResponse || '--', icon: Clock },
   271→            { label: t('profile.stat.avgRating'), value: stats?.avgRating || '--', icon: Shield },
   272→            { label: t('profile.stat.totalConversations'), value: stats ? String(stats.totalConversations) : '--', icon: User },
   273→          ].map((stat) => (
   274→            <div key={stat.label} className="glass-card rounded-xl p-4">
   275→              <stat.icon className="h-4 w-4 text-muted-foreground/40 mb-2" />
   276→              <p className="text-lg font-bold tabular-nums">{stat.value}</p>
   277→              <p className="text-[11px] text-muted-foreground/50 mt-0.5 font-medium">{stat.label}</p>
   278→            </div>
   279→          ))}
   280→        </div>
   281→      </div>
   282→    </div>
   283→  )
   284→}

   286→// ═══ SYSTEM TAB ═══
   287→function SystemTab() {
   288→  const settings = useCRMStore((s) => s.settings)
   289→  const updateSettings = useCRMStore((s) => s.updateSettings)
   290→  const notifCount = useCRMStore((s) => s.notifications.length)
   291→  const clearAllNotifications = useCRMStore((s) => s.clearAllNotifications)
   292→  const { theme, setTheme } = useTheme()
   293→  const [mounted, setMounted] = useState(false)
   294→  const [statusModal, setStatusModal] = useState<{ type: 'success' | 'error'; text: string } | null>(null)
   295→  const { t } = useT()
   296→  useEffect(() => { setMounted(true) }, [])
   297→
   298→  const handleUpdateSetting = async (patch: Record<string, unknown>) => {
   299→    updateSettings(patch)
   300→    try {
   301→      const newSettings = { ...useCRMStore.getState().settings, ...patch }
   302→      await apiPut('/api/auth/me/settings', newSettings)
   303→      setStatusModal({ type: 'success', text: t('settings.saveSuccess') })
   304→    } catch {
   305→      setStatusModal({ type: 'error', text: t('settings.saveFailed') })
   306→    }
   307→  }
   308→
   309→  return (
   310→    <div className="space-y-6 animate-fade-in">
   311→      {/* Status Modal */}
   312→      <Dialog open={!!statusModal} onOpenChange={(open) => { if (!open) setStatusModal(null) }}>
   313→        <DialogContent className="sm:max-w-sm rounded-2xl">
   314→          <DialogHeader>
   315→            <DialogTitle className={cn('text-base flex items-center gap-2', statusModal?.type === 'success' ? 'text-emerald-600' : 'text-red-600')}>
   316→              {statusModal?.type === 'success' ? <Check className="h-5 w-5" /> : <X className="h-5 w-5" />}
   317→              {statusModal?.type === 'success' ? t('common.success') : t('common.failed')}
   318→            </DialogTitle>
   319→            <DialogDescription className="text-sm text-muted-foreground/70">{statusModal?.text}</DialogDescription>
   320→          </DialogHeader>
   321→          <DialogFooter>
   322→            <Button onClick={() => setStatusModal(null)} className="rounded-xl h-9 text-xs">{t('common.close')}</Button>
   323→          </DialogFooter>
   324→        </DialogContent>
   325→      </Dialog>
   326→      <div className="space-y-1">
   327→        <SectionHeader title={t('settings.appearance')} />
   328→        <div className="glass-card rounded-2xl p-5">
   329→          <SettingRow icon={mounted && theme === 'dark' ? Moon : Sun} label={t('settings.theme')} description={t('settings.themeDesc')}>
   330→            <Button variant="outline" size="sm" className="h-8 rounded-lg text-xs gap-1.5 border-border/40" onClick={() => setTheme(theme === 'dark' ? 'light' : 'dark')}>
   331→              {mounted && (theme === 'dark' ? <Moon className="h-3.5 w-3.5" /> : <Sun className="h-3.5 w-3.5" />)}
   332→              {mounted ? (theme === 'dark' ? t('settings.dark') : t('settings.light')) : ''}
   333→            </Button>
   334→          </SettingRow>
   335→          <Separator className="opacity-30 my-1" />
   336→          <SettingRow icon={Maximize2} label={t('settings.compactMode')} description={t('settings.compactModeDesc')}>
   337→            <Switch checked={settings.compactMode} onCheckedChange={(v) => handleUpdateSetting({ compactMode: v })} />
   338→          </SettingRow>
   339→          <Separator className="opacity-30 my-1" />
   340→          <SettingRow icon={Eye} label={t('settings.showPreview')} description={t('settings.showPreviewDesc')}>
   341→            <Switch checked={settings.showPreview} onCheckedChange={(v) => handleUpdateSetting({ showPreview: v })} />
   342→          </SettingRow>
   343→          <Separator className="opacity-30 my-1" />
   344→          <SettingRow icon={Globe} label={t('settings.language')}>
   345→            <Select value={settings.language} onValueChange={(v) => handleUpdateSetting({ language: v as Locale })}>
   346→              <SelectTrigger className="w-32 h-8 rounded-lg text-xs"><SelectValue /></SelectTrigger>
   347→              <SelectContent>{LOCALES.map((loc) => <SelectItem key={loc} value={loc}>{LOCALE_LABELS[loc]}</SelectItem>)}</SelectContent>
   348→            </Select>
   349→          </SettingRow>
   350→        </div>
   351→      </div>
   352→      <div className="space-y-1">
   353→        <SectionHeader title={t('settings.notifications')} />
   354→        <div className="glass-card rounded-2xl p-5">
   355→          <SettingRow icon={Volume2} label={t('settings.sound')} description={t('settings.soundDesc')}>
   356→            <Switch checked={settings.soundEnabled} onCheckedChange={(v) => handleUpdateSetting({ soundEnabled: v })} />
   357→          </SettingRow>
   358→          <Separator className="opacity-30 my-1" />
   359→          <SettingRow icon={Monitor} label={t('settings.desktopNotif')} description={t('settings.desktopNotifDesc')}>
   360→            <Switch checked={settings.desktopNotifEnabled} onCheckedChange={(v) => handleUpdateSetting({ desktopNotifEnabled: v })} />
   361→          </SettingRow>
   362→          <Separator className="opacity-30 my-1" />
   363→          <SettingRow icon={Mail} label={t('settings.emailNotif')} description={t('settings.emailNotifDesc')}>
   364→            <Switch checked={settings.emailNotifEnabled} onCheckedChange={(v) => handleUpdateSetting({ emailNotifEnabled: v })} />
   365→          </SettingRow>
   366→          {notifCount > 0 && (
   367→            <><Separator className="opacity-30 my-1" />
   368→            <Button variant="ghost" size="sm" className="w-full mt-1 h-9 rounded-xl text-xs text-muted-foreground/60 hover:text-foreground hover:bg-foreground/[0.03]" onClick={clearAllNotifications}>
   369→              <Trash2 className="h-3.5 w-3.5 mr-1.5" /> {t('settings.clearAllNotifs', { count: notifCount })}
   370→            </Button></>
   371→          )}
   372→        </div>
   373→      </div>
   374→      <div className="space-y-1">
   375→        <SectionHeader title={t('settings.conversations')} />
   376→        <div className="glass-card rounded-2xl p-5">
   377→          <SettingRow icon={UserCheck} label={t('settings.autoAssign')} description={t('settings.autoAssignDesc')}>
   378→            <Switch checked={settings.autoAssign} onCheckedChange={(v) => handleUpdateSetting({ autoAssign: v })} />
   379→          </SettingRow>
   380→        </div>
   381→      </div>
   382→      <div className="space-y-1">
   383→        <SectionHeader title={t('settings.data')} />
   384→        <div className="glass-card rounded-2xl p-5">
   385→          <SettingRow icon={Trash2} label={t('settings.clearCache')} description={t('settings.clearCacheDesc')}>
   386→            <Button variant="outline" size="sm" className="h-8 rounded-lg text-xs border-border/40 hover:border-red-300 hover:text-red-600">{t('settings.clearCache')}</Button>
   387→          </SettingRow>
   388→          <Separator className="opacity-30 my-1" />
   389→          <Button variant="outline" size="sm" className="w-full h-9 rounded-xl text-xs border-border/40 hover:border-red-300 hover:text-red-600" onClick={() => { localStorage.removeItem('omnichat_settings'); updateSettings({ soundEnabled: true, desktopNotifEnabled: true, emailNotifEnabled: false, compactMode: false, showPreview: true, autoAssign: true, language: settings.language }) }}>
   390→            <RotateCcw className="h-3.5 w-3.5 mr-1.5" /> {t('settings.resetDefaults')}
   391→          </Button>
   392→        </div>
   393→      </div>
   394→    </div>
   395→  )

   609→function StaffTab() {
   610→  const agents = useCRMStore((s) => s.agents)
   611→  const setAgents = useCRMStore((s) => s.setAgents)
   612→  const { t } = useT()
   613→  const [loading, setLoading] = useState(true)
   614→  const [showInvite, setShowInvite] = useState(false)
   615→  const [inviteEmail, setInviteEmail] = useState('')
   616→  const [inviteRole, setInviteRole] = useState('agent')
   617→  const [editingAgent, setEditingAgent] = useState<Agent | null>(null)
   618→  const [expandedId, setExpandedId] = useState<string | null>(null)
   619→  const [deleteConfirm, setDeleteConfirm] = useState(false)
   620→  const [editForm, setEditForm] = useState({ name: '', email: '', role: 'agent', status: 'online' })
   621→
   622→  // Generate stable mock stats per agent
   623→  const staffStats = useMemo(() => {
   624→    const stats: Record<string, { conversations: number; avgResponse: number; satisfaction: number; lastActive: string }> = {}
   625→    agents.forEach((agent) => {
   626→      // Simple deterministic hash from agent id
   627→      const hash = agent.id.split('').reduce((a, c) => a + c.charCodeAt(0), 0)
   628→      const conversations = 100 + (hash * 7) % 500
   629→      const avgResponse = 60 + (hash * 13) % 120 // 1-3 min in seconds
   630→      const satisfaction = 4.0 + ((hash * 3) % 10) / 10 // 4.0-5.0
   631→      const daysAgo = (hash * 11) % 30
   632→      const hoursAgo = (hash * 7) % 24
   633→      const lastActive = daysAgo === 0
   634→        ? t('common.time.hoursAgo', { h: hoursAgo })
   635→        : t('common.time.daysAgo', { d: daysAgo })
   636→      stats[agent.id] = { conversations, avgResponse, satisfaction, lastActive }
   637→    })
   638→    return stats
   639→  }, [agents])
   640→
   641→  useEffect(() => {
   642→    if (agents.length > 0) { setLoading(false); return }
   643→    async function load() {
   644→      try {
   645→        const res = await fetch('/api/agents')
   646→        const data = await res.json()
   647→        if (Array.isArray(data)) { setAgents(data) }
   648→      } catch (e) { console.error('Failed', e) }
   649→      finally { setLoading(false) }
   650→    }
   651→    load()
   652→  }, [agents.length, setAgents])
   653→
   654→  const handleInvite = () => {
   655→    if (!inviteEmail.trim()) return
   656→    const newAgent: Agent = { id: `agent_${Date.now()}`, name: inviteEmail.split('@')[0], email: inviteEmail, avatar: null, role: inviteRole, status: 'offline' }
   657→    setAgents([...agents, newAgent])
   658→    setInviteEmail('')
   659→    setShowInvite(false)
   660→  }
   661→
   662→  const openEditDialog = (agent: Agent, e?: React.MouseEvent) => {
   663→    if (e) e.stopPropagation()
   664→    setEditForm({ name: agent.name, email: agent.email, role: agent.role, status: agent.status })
   665→    setDeleteConfirm(false)
   666→    setEditingAgent(agent)
   667→  }
   668→
   669→  const handleSaveEdit = () => {
   670→    if (!editingAgent) return
   671→    const updated = { ...editingAgent, ...editForm }
   672→    const newList = agents.map(a => a.id === editingAgent.id ? updated : a)
   673→    setAgents(newList)
   674→    setEditingAgent(null)
   675→  }
   676→
   677→  const handleDelete = () => {
   678→    if (!editingAgent) return
   679→    const newList = agents.filter(a => a.id !== editingAgent.id)
   680→    setAgents(newList)
   681→    setEditingAgent(null)
   682→    setDeleteConfirm(false)
   683→  }
   684→
   685→  const toggleExpand = (id: string) => {

   687→  }
   688→

   690→  const roleBadge = (role: string) => role === 'admin' ? 'bg-indigo-100 text-indigo-700 dark:bg-indigo-950/40 dark:text-indigo-400' : role === 'supervisor' ? 'bg-violet-100 text-violet-700 dark:bg-violet-950/40 dark:text-violet-400' : 'bg-slate-100 text-slate-600 dark:bg-slate-800/40 dark:text-slate-400'
   691→

   694→    const sec = seconds % 60
   695→    return min > 0 ? `~${min}m ${sec > 0 ? `${sec}s` : ''}` : `~${sec}s`

  1161→function BackupTab() {

  1164→  const [loading, setLoading] = useState(true)

  1168→  const [message, setMessage] = useState<{ type: 'success' | 'error'; text: string } | null>(null)
  1169→  const [confirmAction, setConfirmAction] = useState<{ type: 'delete' | 'restore'; id: string } | null>(null)
  1170→
  1171→  const loadBackups = async () => {
  1172→    setLoading(true)
  1173→    try {
  1174→      const res = await fetch('/api/backup')
  1175→      const data = await res.json()
  1176→      setBackups(data.data || [])
  1177→    } catch (e) { console.error(e) }
  1178→    finally { setLoading(false) }
  1179→  }
  1180→
  1181→  useEffect(() => { loadBackups() }, [])
  1182→
  1183→  const createBackup = async () => {
  1184→    setCreating(true)
  1185→    setMessage(null)
  1186→    try {
  1187→      const data: any = await apiPost('/api/backup', {})
  1188→      setMessage({ type: 'success', text: t('backup.created') })
  1189→      loadBackups()
  1190→    } catch (e: any) {
  1191→      setMessage({ type: 'error', text: e?.data?.error || t('backup.createFailed') })
  1192→    }
  1193→    finally { setCreating(false) }
  1194→  }
  1195→
  1196→  const restoreBackup = async (id: string) => {
  1197→    setRestoring(id)
  1198→    setMessage(null)
  1199→    try {
  1200→      await apiPost('/api/backup/restore', { backupId: id })
  1201→      setMessage({ type: 'success', text: t('backup.restored') })
  1202→    } catch (e: any) {
  1203→      setMessage({ type: 'error', text: e?.data?.error || t('backup.restoreFailed') })
  1204→    }
  1205→    finally { setRestoring(null) }
  1206→  }
  1207→
  1208→  const deleteBackup = async (id: string) => {
  1209→    setDeleting(id)
  1210→    try {
  1211→      await apiFetch(`/api/backup?id=${id}`, { method: 'DELETE' })
  1212→      setBackups(prev => prev.filter(b => b.id !== id))
  1213→      setMessage({ type: 'success', text: t('backup.deleteSuccess') })
  1214→    } catch {
  1215→      setMessage({ type: 'error', text: t('backup.deleteFailed') })
  1216→    }
  1217→    finally { setDeleting(null) }
  1218→  }
  1219→
  1220→  const executeConfirmedAction = async () => {
  1221→    if (!confirmAction) return
  1222→    if (confirmAction.type === 'restore') {
  1223→      setConfirmAction(null)
  1224→      await restoreBackup(confirmAction.id)
  1225→    } else {
  1226→      setConfirmAction(null)
  1227→      await deleteBackup(confirmAction.id)
  1228→    }
  1229→  }
  1230→
  1231→  const formatSize = (bytes: number) => {
  1232→    if (bytes < 1024) return `${bytes} B`
  1233→    if (bytes < 1048576) return `${(bytes / 1024).toFixed(1)} KB`
  1234→    return `${(bytes / 1048576).toFixed(1)} MB`
  1235→  }
  1236→
  1237→  if (loading) {
  1238→    return (
  1239→      <div className="flex items-center justify-center py-20">
  1240→        <Loader2 className="h-6 w-6 animate-spin text-muted-foreground/40" />
  1241→      </div>
  1242→    )
  1243→  }
  1244→
  1245→  return (
  1246→    <div className="space-y-6 animate-fade-in">
  1247→      <div className="space-y-1">
  1248→        <SectionHeader title={t('backup.manage')} />
  1249→        <div className="glass-card rounded-2xl p-5 space-y-4">
  1250→          {/* Info */}
