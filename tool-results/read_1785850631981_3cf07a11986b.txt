   137→   137→function ProfileTab() {
   138→   138→  const currentUser = useCRMStore((s) => s.currentUser)
   139→   139→  const setCurrentUser = useCRMStore((s) => s.setCurrentUser)
   140→   140→  const { t } = useT()
   141→   141→  const [editing, setEditing] = useState(false)
   142→   142→  const [saving, setSaving] = useState(false)
   143→   143→  const [form, setForm] = useState({ name: '', email: '', phone: '', bio: '' })
   144→   144→  const fileRef = useRef<HTMLInputElement>(null)
   145→   145→  const [stats, setStats] = useState<AgentStats | null>(null)
   146→   146→
   147→   147→  useEffect(() => {
   148→   148→    fetch('/api/agents/me/stats')
   149→   149→      .then(res => res.ok ? res.json() : null)
   150→   150→      .then(data => { if (data) setStats(data) })
   151→   151→      .catch(() => {})
   152→   152→  }, [])
   153→   153→
   154→   154→  useEffect(() => {
   155→   155→    if (currentUser) setForm({ name: currentUser.name, email: currentUser.email, phone: currentUser.phone, bio: currentUser.bio })
   156→   156→  }, [currentUser])
   157→   157→
   158→   158→  const handleSave = async () => {
   159→   159→    if (!currentUser) return
   160→   160→    setSaving(true)
   161→   161→    try {
   162→   162→      const updated = await apiPut('/api/auth/me', { name: form.name, email: form.email, phone: form.phone, bio: form.bio })
   163→   163→      setCurrentUser({ ...currentUser, ...updated })
   164→   164→      setEditing(false)
   165→   165→    } catch { /* keep local state on error */ }
   166→   166→    setSaving(false)
   167→   167→  }
   168→   168→  const handleCancel = () => { if (currentUser) setForm({ name: currentUser.name, email: currentUser.email, phone: currentUser.phone, bio: currentUser.bio }); setEditing(false) }
   169→   169→  const handleStatusChange = async (status: UserProfile['status']) => {
   170→   170→    if (!currentUser) return
   171→   171→    setCurrentUser({ ...currentUser, status })
   172→   172→    try { await apiPut('/api/auth/me', { status }) } catch { /* optimistic update already applied */ }
   173→   173→  }
   174→   174→
   175→   175→  if (!currentUser) return null
   176→   176→
   177→   177→  return (
   178→   178→    <div className="space-y-6 animate-fade-in">
   179→   179→      <div className="glass-card rounded-2xl p-6">
   180→   180→        <div className="flex flex-col items-center gap-4">
   181→   181→          <div className="relative group">
   182→   182→            <Avatar className={cn('h-24 w-24 ring-4 ring-background shadow-xl', GRADIENT_CLASSES[0])}>
   183→   183→              <AvatarFallback className="text-3xl text-white font-bold">{currentUser.name.split(' ').slice(-2).map(n => n[0]).join('')}</AvatarFallback>
   184→   184→            </Avatar>
   185→   185→            <button onClick={() => fileRef.current?.click()} className="absolute inset-0 rounded-full bg-black/0 group-hover:bg-black/30 transition-all duration-200 flex items-center justify-center opacity-0 group-hover:opacity-100">
   186→   186→              <Camera className="h-6 w-6 text-white" />
   187→   187→            </button>
   188→   188→            <input ref={fileRef} type="file" accept="image/*" className="hidden" onChange={() => {}} />
   189→   189→          </div>
   190→   190→          {!editing && (
   191→   191→            <div className="text-center">
   192→   192→              <h3 className="text-xl font-bold tracking-tight">{currentUser.name}</h3>
   193→   193→              <p className="text-sm text-muted-foreground/60 mt-1 font-medium">{currentUser.email}</p>
   194→   194→              <span className="inline-flex items-center gap-1.5 mt-2.5 text-xs font-medium px-3 py-1.5 rounded-full bg-foreground/[0.04] text-muted-foreground/70">
   195→   195→                <Shield className="h-3.5 w-3.5" />
   196→   196→                {currentUser.role === 'admin' ? t('user.role.admin') : currentUser.role === 'agent' ? t('user.role.agent') : currentUser.role === 'supervisor' ? t('user.role.supervisor') : currentUser.role}
   197→   197→              </span>
   198→   198→            </div>
   199→   199→          )}
   200→   200→          <div className="flex gap-2">
   201→   201→            {STATUS_OPTIONS.map((opt) => (
   202→   202→              <button key={opt.value} onClick={() => handleStatusChange(opt.value)} className={cn('flex items-center gap-1.5 px-3.5 py-2 rounded-xl text-xs font-medium transition-all duration-200 border', currentUser.status === opt.value ? 'border-primary/30 bg-primary/5 text-primary' : 'border-transparent text-muted-foreground/50 hover:text-foreground hover:bg-foreground/[0.03]')}>
   203→   203→                <span className={cn('h-2 w-2 rounded-full', opt.color, currentUser.status === opt.value && 'shadow-sm')} />
   204→   204→                {t(opt.labelKey)}
   205→   205→              </button>
   206→   206→            ))}
   207→   207→          </div>
   208→   208→        </div>
   209→   209→      </div>
   210→   210→      <div className="space-y-4">
   211→   211→        <SectionHeader title={t('profile.personalInfo')} />
   212→   212→        <div className="glass-card rounded-2xl p-5 space-y-4">
   213→   213→          <div className="space-y-3">
   214→   214→            <div className="space-y-1.5">
   215→   215→              <Label className="text-xs font-medium text-muted-foreground/70">{t('profile.displayName')}</Label>
   216→   216→              <Input value={form.name} onChange={(e) => setForm(f => ({ ...f, name: e.target.value }))} disabled={!editing} className="rounded-xl glass-input h-10 text-sm" />
   217→   217→            </div>
   218→   218→            <div className="space-y-1.5">
   219→   219→              <Label className="text-xs font-medium text-muted-foreground/70">{t('profile.email')}</Label>
   220→   220→              <Input type="email" value={form.email} onChange={(e) => setForm(f => ({ ...f, email: e.target.value }))} disabled={!editing} className="rounded-xl glass-input h-10 text-sm" />
   221→   221→            </div>
   222→   222→            <div className="space-y-1.5">
   223→   223→              <Label className="text-xs font-medium text-muted-foreground/70">{t('profile.phoneNumber')}</Label>
   224→   224→              <Input value={form.phone} onChange={(e) => setForm(f => ({ ...f, phone: e.target.value }))} disabled={!editing} className="rounded-xl glass-input h-10 text-sm" />
   225→   225→            </div>
   226→   226→            <div className="space-y-1.5">
   227→   227→              <Label className="text-xs font-medium text-muted-foreground/70">{t('profile.bio')}</Label>
   228→   228→              <Textarea value={form.bio} onChange={(e) => setForm(f => ({ ...f, bio: e.target.value }))} disabled={!editing} className="rounded-xl glass-input min-h-[80px] resize-none text-sm" />
   229→   229→            </div>
   230→   230→          </div>
   231→   231→          {!editing ? (
   232→   232→            <Button onClick={() => setEditing(true)} className="w-full h-10 rounded-xl text-sm font-medium"><Pencil className="h-4 w-4 mr-1.5" /> {t('profile.editInfo')}</Button>
   233→   233→          ) : (
   234→   234→            <div className="flex gap-2">
   235→   235→              <Button variant="outline" onClick={handleCancel} className="flex-1 h-10 rounded-xl text-sm font-medium"><X className="h-4 w-4 mr-1.5" /> {t('profile.cancel')}</Button>
   236→   236→              <Button onClick={handleSave} disabled={saving} className="flex-1 h-10 rounded-xl text-sm font-medium bg-gradient-to-r from-indigo-500 to-violet-500 hover:from-indigo-600 hover:to-violet-600">{saving ? <Loader2 className="h-4 w-4 mr-1.5 animate-spin" /> : <Check className="h-4 w-4 mr-1.5" />} {t('profile.save')}</Button>
   237→   237→            </div>
   238→   238→          )}
   239→   239→        </div>
   240→   240→      </div>
   241→   241→      <div className="space-y-2">
   242→   242→        <SectionHeader title={t('profile.stats.title')} />
   243→   243→        <div className="grid grid-cols-2 gap-3">
   244→   244→          {[
   245→   245→            { label: t('profile.stat.conversationsToday'), value: stats ? String(stats.conversationsToday) : '--', icon: MessageSquare },
   246→   246→            { label: t('profile.stat.avgResponse'), value: stats?.avgResponse || '--', icon: Clock },
   247→   247→            { label: t('profile.stat.avgRating'), value: stats?.avgRating || '--', icon: Shield },
   248→   248→            { label: t('profile.stat.totalConversations'), value: stats ? String(stats.totalConversations) : '--', icon: User },
   249→   249→          ].map((stat) => (
   250→   250→            <div key={stat.label} className="glass-card rounded-xl p-4">
   251→   251→              <stat.icon className="h-4 w-4 text-muted-foreground/40 mb-2" />
   252→   252→              <p className="text-lg font-bold tabular-nums">{stat.value}</p>
   253→   253→              <p className="text-[11px] text-muted-foreground/50 mt-0.5 font-medium">{stat.label}</p>
   254→   254→            </div>
   255→   255→          ))}
   256→   256→        </div>
   257→   257→      </div>
   258→   258→    </div>
   259→   259→  )
   260→   260→}

   557→   557→function StaffTab() {

   560→   560→  const { t } = useT()

   567→   567→  const [deleteConfirm, setDeleteConfirm] = useState(false)
   568→   568→  const [editForm, setEditForm] = useState({ name: '', email: '', role: 'agent', status: 'online' })
   569→   569→
   570→   570→  // Generate stable mock stats per agent
   571→   571→  const staffStats = useMemo(() => {
   572→   572→    const stats: Record<string, { conversations: number; avgResponse: number; satisfaction: number; lastActive: string }> = {}
   573→   573→    agents.forEach((agent) => {
   574→   574→      // Simple deterministic hash from agent id
   575→   575→      const hash = agent.id.split('').reduce((a, c) => a + c.charCodeAt(0), 0)
   576→   576→      const conversations = 100 + (hash * 7) % 500
   577→   577→      const avgResponse = 60 + (hash * 13) % 120 // 1-3 min in seconds
   578→   578→      const satisfaction = 4.0 + ((hash * 3) % 10) / 10 // 4.0-5.0
   579→   579→      const daysAgo = (hash * 11) % 30
   580→   580→      const hoursAgo = (hash * 7) % 24
   581→   581→      const lastActive = daysAgo === 0
   582→   582→        ? t('common.time.hoursAgo', { h: hoursAgo })
   583→   583→        : t('common.time.daysAgo', { d: daysAgo })
   584→   584→      stats[agent.id] = { conversations, avgResponse, satisfaction, lastActive }
   585→   585→    })
   586→   586→    return stats
   587→   587→  }, [agents])
   588→   588→
   589→   589→  useEffect(() => {
   590→   590→    if (agents.length > 0) { setLoading(false); return }
   591→   591→    async function load() {
   592→   592→      try {
   593→   593→        const res = await fetch('/api/agents')
   594→   594→        const data = await res.json()
   595→   595→        if (Array.isArray(data)) { setAgents(data) }
   596→   596→      } catch (e) { console.error('Failed', e) }
