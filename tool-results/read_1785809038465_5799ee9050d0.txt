    54→function shouldShowDate(messages: Message[], idx: number) {
    55→  if (idx === 0) return true
    56→  const prev = new Date(messages[idx - 1].createdAt).toDateString()
    57→  const curr = new Date(messages[idx].createdAt).toDateString()
    58→  return prev !== curr
    59→}

   314→        requestAnimationFrame(() => {
   315→          if (container) {
   316→            const newScrollHeight = container.scrollHeight
   317→            container.scrollTop = newScrollHeight - prevScrollHeight + prevScrollTop
   318→          }
   319→        })
   320→      } else {
   321→        setHasMoreMessages(false)
   322→      }
   323→    } catch (e) {
   324→      console.error('Failed to load older messages', e)
   325→    } finally {
   326→      loadingMoreRef.current = false
   327→      setIsLoadingMoreMessages(false)
