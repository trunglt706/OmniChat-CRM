   142→  const [form, setForm] = useState({ name: '', email: '', phone: '', bio: '' })
   143→  const fileRef = useRef<HTMLInputElement>(null)
   144→  const [stats, setStats] = useState<AgentStats | null>(null)
   145→
   146→  useEffect(() => {
   147→    fetch('/api/agents/me/stats')
   148→      .then(res => res.ok ? res.json() : null)
   149→      .then(data => { if (data) setStats(data) })
   150→      .catch(() => {})
   151→  }, [])
   152→
   153→  useEffect(() => {
   154→    if (currentUser) setForm({ name: currentUser.name, email: currentUser.email, phone: currentUser.phone, bio: currentUser.bio })
   155→  }, [currentUser])

   158→    if (!currentUser) return
   159→    setSaving(true)
   160→    try {
   161→      const updated = await apiPut('/api/auth/me', { name: form.name, email: form.email, phone: form.phone, bio: form.bio })
   162→      setCurrentUser({ ...currentUser, ...updated })
   163→      setEditing(false)
   164→    } catch { /* keep local state on error */ }

   340→// ═══ CHANNELS TAB ═══
   341→function ChannelsTab() {
   342→  const { t } = useT()
   343→  const [channels, setChannels] = useState<ChannelData[]>([])
   344→  const [loading, setLoading] = useState(true)
   345→  const [editingChannel, setEditingChannel] = useState<string | null>(null)
   346→  const [editForm, setEditForm] = useState<Record<string, string>>({})
   347→  const [saving, setSaving] = useState(false)
   348→  const [testingKey, setTestingKey] = useState<string | null>(null)
   349→  const [testResults, setTestResults] = useState<Record<string, { ok: boolean; msg: string } | null>>({})
   350→
   351→  // Load channels from API
   352→  useEffect(() => {
   353→    fetch('/api/channels')
   354→      .then(r => r.json())
   355→      .then(data => { if (Array.isArray(data)) setChannels(data) })
   356→      .catch(() => {})
   357→      .finally(() => setLoading(false))
   358→  }, [])
   359→
   360→  const toggleChannel = async (key: string) => {
   361→    const ch = channels.find(c => c.key === key)
   362→    if (!ch) return
   363→    setChannels(prev => prev.map(c => c.key === key ? { ...c, enabled: !c.enabled } : c))
   364→    await fetch('/api/channels', {
   365→      method: 'PUT',
   366→      headers: { 'Content-Type': 'application/json' },
   367→      body: JSON.stringify({ channel: key, enabled: !ch.enabled, config: ch.config }),
   368→    }).catch(() => {})
   369→  }
   370→
   371→  const startEdit = (ch: ChannelData) => {
   372→    setEditingChannel(ch.key)
   373→    setEditForm({ ...ch.config })
   374→    setTestResults(prev => ({ ...prev, [ch.key]: null }))
   375→  }
   376→
   377→  const saveChannel = async (key: string) => {
   378→    setSaving(true)
   379→    try {
   380→      const res = await fetch('/api/channels', {
   381→        method: 'PUT',
   382→        headers: { 'Content-Type': 'application/json' },
   383→        body: JSON.stringify({ channel: key, config: editForm }),
   384→      })
   385→      const data = await res.json()
   386→      setChannels(prev => prev.map(c => c.key === key ? { ...c, configured: data.configured, config: data.config || c.config } : c))
   387→      setEditingChannel(null)
   388→    } catch {}
   389→    setSaving(false)
   390→  }
   391→
   392→  const testConnection = async (key: string) => {
   393→    setTestingKey(key)
   394→    setTestResults(prev => ({ ...prev, [key]: null }))
   395→    try {
   396→      const ch = channels.find(c => c.key === key)
   397→      const res = await fetch('/api/channels/test', {
   398→        method: 'POST',
   399→        headers: { 'Content-Type': 'application/json' },
   400→        body: JSON.stringify({ channel: key, config: { ...ch?.config, ...editForm } }),
   401→      })
   402→      const result = await res.json()
   403→      setTestResults(prev => ({ ...prev, [key]: result }))
   404→      // Update channel data to reflect new test result
   405→      setChannels(prev => prev.map(c => c.key === key ? {
   406→        ...c,
   407→        lastTestAt: new Date().toISOString(),
   408→        lastTestOk: result.ok,
   409→        lastTestMsg: result.msg,
   410→      } : c))
   411→    } catch {
   412→      setTestResults(prev => ({ ...prev, [key]: { ok: false, msg: t('common.error.serverConnection') } }))
   413→    }
   414→    setTestingKey(null)
