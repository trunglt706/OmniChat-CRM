    52→// ═══ Shared fetch cache (prevents duplicate API calls across mounts) ═══
    53→const fetchCache = new Map<string, { promise: Promise<any>; ts: number }>()
    54→const CACHE_TTL = 5000
    55→function cachedFetch(url: string): Promise<any> {
    56→  const now = Date.now()
    57→  const cached = fetchCache.get(url)
    58→  if (cached && now - cached.ts < CACHE_TTL) return cached.promise
    59→  const promise = fetch(url).then(r => r.ok ? r.json() : null)
    60→  fetchCache.set(url, { promise, ts: now })
    61→  return promise
    62→}

   163→

   165→    cachedFetch('/api/agents/me/stats')
   166→      .then(data => { if (data) setStats(data) })

   171→    if (currentUser) setForm({ name: currentUser.name, email: currentUser.email, phone: currentUser.phone, bio: currentUser.bio })

   622→function StaffTab() {
   623→  const agents = useCRMStore((s) => s.agents)
   624→  const setAgents = useCRMStore((s) => s.setAgents)
   625→  const { t } = useT()
   626→  const [loading, setLoading] = useState(true)
   627→  const [showInvite, setShowInvite] = useState(false)
   628→  const [inviteEmail, setInviteEmail] = useState('')
   629→  const [inviteRole, setInviteRole] = useState('agent')
   630→  const [editingAgent, setEditingAgent] = useState<Agent | null>(null)
   631→  const [expandedId, setExpandedId] = useState<string | null>(null)
   632→  const [deleteConfirm, setDeleteConfirm] = useState(false)
   633→  const [editForm, setEditForm] = useState({ name: '', email: '', role: 'agent', status: 'online' })
   634→
   635→  // Generate stable mock stats per agent
   636→  const staffStats = useMemo(() => {
   637→    const stats: Record<string, { conversations: number; avgResponse: number; satisfaction: number; lastActive: string }> = {}
   638→    agents.forEach((agent) => {
   639→      // Simple deterministic hash from agent id
   640→      const hash = agent.id.split('').reduce((a, c) => a + c.charCodeAt(0), 0)
   641→      const conversations = 100 + (hash * 7) % 500
   642→      const avgResponse = 60 + (hash * 13) % 120 // 1-3 min in seconds
   643→      const satisfaction = 4.0 + ((hash * 3) % 10) / 10 // 4.0-5.0
   644→      const daysAgo = (hash * 11) % 30
   645→      const hoursAgo = (hash * 7) % 24
   646→      const lastActive = daysAgo === 0
   647→        ? t('common.time.hoursAgo', { h: hoursAgo })
   648→        : t('common.time.daysAgo', { d: daysAgo })
   649→      stats[agent.id] = { conversations, avgResponse, satisfaction, lastActive }
   650→    })
   651→    return stats
   652→  }, [agents])
   653→
   654→  useEffect(() => {
   655→    if (agents.length > 0) { setLoading(false); return }
   656→    async function load() {
   657→      try {
   658→        const data = await cachedFetch('/api/agents')
   659→        if (Array.isArray(data)) { setAgents(data) }
   660→      } catch (e) { console.error('Failed', e) }
   661→      finally { setLoading(false) }
   662→    }
   663→    load()
   664→  }, [agents.length, setAgents])
   665→
   666→  const handleInvite = () => {
   667→    if (!inviteEmail.trim()) return
   668→    const newAgent: Agent = { id: `agent_${Date.now()}`, name: inviteEmail.split('@')[0], email: inviteEmail, avatar: null, role: inviteRole, status: 'offline' }
   669→    setAgents([...agents, newAgent])
   670→    setInviteEmail('')
   671→    setShowInvite(false)
   672→  }
   673→
   674→  const openEditDialog = (agent: Agent, e?: React.MouseEvent) => {
   675→    if (e) e.stopPropagation()
   676→    setEditForm({ name: agent.name, email: agent.email, role: agent.role, status: agent.status })
   677→    setDeleteConfirm(false)
   678→    setEditingAgent(agent)
   679→  }
   680→
   681→  const handleSaveEdit = () => {
   682→    if (!editingAgent) return
   683→    const updated = { ...editingAgent, ...editForm }
   684→    const newList = agents.map(a => a.id === editingAgent.id ? updated : a)
   685→    setAgents(newList)
   686→    setEditingAgent(null)
   687→  }
   688→
   689→  const handleDelete = () => {
   690→    if (!editingAgent) return
   691→    const newList = agents.filter(a => a.id !== editingAgent.id)
   692→    setAgents(newList)
   693→    setEditingAgent(null)
   694→    setDeleteConfirm(false)
   695→  }
   696→
   697→  const toggleExpand = (id: string) => {
   698→    setExpandedId(prev => prev === id ? null : id)
   699→  }
   700→
   701→  const roleLabel = (role: string) => role === 'admin' ? t('user.role.admin') : role === 'supervisor' ? t('user.role.supervisor') : role === 'agent' ? t('user.role.agent') : role
   702→  const roleBadge = (role: string) => role === 'admin' ? 'bg-indigo-100 text-indigo-700 dark:bg-indigo-950/40 dark:text-indigo-400' : role === 'supervisor' ? 'bg-violet-100 text-violet-700 dark:bg-violet-950/40 dark:text-violet-400' : 'bg-slate-100 text-slate-600 dark:bg-slate-800/40 dark:text-slate-400'
   703→
   704→  const formatResponseTime = (seconds: number) => {
   705→    const min = Math.floor(seconds / 60)
   706→    const sec = seconds % 60
   707→    return min > 0 ? `~${min}m ${sec > 0 ? `${sec}s` : ''}` : `~${sec}s`
   708→  }
   709→

   711→    <div className="space-y-6 animate-fade-in">

  1166→interface BackupItem {

  1172→
  1173→function BackupTab() {
  1174→  const { t } = useT()
  1175→  const [backups, setBackups] = useState<BackupItem[]>([])
  1176→  const [loading, setLoading] = useState(true)
  1177→  const [creating, setCreating] = useState(false)
  1178→  const [restoring, setRestoring] = useState<string | null>(null)
  1179→  const [deleting, setDeleting] = useState<string | null>(null)
  1180→  const [message, setMessage] = useState<{ type: 'success' | 'error'; text: string } | null>(null)
  1181→  const [confirmAction, setConfirmAction] = useState<{ type: 'delete' | 'restore'; id: string } | null>(null)
  1182→
  1183→  const loadBackups = async () => {
  1184→    setLoading(true)
  1185→    try {
  1186→      const res = await fetch('/api/backup')
  1187→      const data = await res.json()
  1188→      setBackups(data.data || [])
  1189→    } catch (e) { console.error(e) }
  1190→    finally { setLoading(false) }
  1191→  }
  1192→
  1193→  useEffect(() => { loadBackups() }, [])
  1194→
  1195→  const createBackup = async () => {
  1196→    setCreating(true)
  1197→    setMessage(null)
  1198→    try {
  1199→      const data: any = await apiPost('/api/backup', {})
  1200→      setMessage({ type: 'success', text: t('backup.created') })
  1201→      loadBackups()
  1202→    } catch (e: any) {
  1203→      setMessage({ type: 'error', text: e?.data?.error || t('backup.createFailed') })
  1204→    }
  1205→    finally { setCreating(false) }
  1206→  }
  1207→
  1208→  const restoreBackup = async (id: string) => {
  1209→    setRestoring(id)
  1210→    setMessage(null)
  1211→    try {
  1212→      await apiPost('/api/backup/restore', { backupId: id })
  1213→      setMessage({ type: 'success', text: t('backup.restored') })
  1214→    } catch (e: any) {
  1215→      setMessage({ type: 'error', text: e?.data?.error || t('backup.restoreFailed') })
  1216→    }
  1217→    finally { setRestoring(null) }
  1218→  }
  1219→
  1220→  const deleteBackup = async (id: string) => {
  1221→    setDeleting(id)
  1222→    try {
  1223→      await apiFetch(`/api/backup?id=${id}`, { method: 'DELETE' })
  1224→      setBackups(prev => prev.filter(b => b.id !== id))
  1225→      setMessage({ type: 'success', text: t('backup.deleteSuccess') })
  1226→    } catch {
  1227→      setMessage({ type: 'error', text: t('backup.deleteFailed') })
  1228→    }
  1229→    finally { setDeleting(null) }
  1230→  }
  1231→
  1232→  const executeConfirmedAction = async () => {
  1233→    if (!confirmAction) return
  1234→    if (confirmAction.type === 'restore') {
  1235→      setConfirmAction(null)
  1236→      await restoreBackup(confirmAction.id)
  1237→    } else {
  1238→      setConfirmAction(null)
  1239→      await deleteBackup(confirmAction.id)
  1240→    }
  1241→  }
  1242→
  1243→  const formatSize = (bytes: number) => {
  1244→    if (bytes < 1024) return `${bytes} B`
  1245→    if (bytes < 1048576) return `${(bytes / 1024).toFixed(1)} KB`
  1246→    return `${(bytes / 1048576).toFixed(1)} MB`
  1247→  }
  1248→
  1249→  if (loading) {
