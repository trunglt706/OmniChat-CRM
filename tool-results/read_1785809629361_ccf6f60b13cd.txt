   345→function ChannelsTab() {
   346→  const { t } = useT()
   347→  const [channels, setChannels] = useState<ChannelData[]>([])
   348→  const [loading, setLoading] = useState(true)
   349→  const [editingChannel, setEditingChannel] = useState<string | null>(null)
   350→  const [editForm, setEditForm] = useState<Record<string, string>>({})
   351→  const [saving, setSaving] = useState(false)
   352→  const [testingKey, setTestingKey] = useState<string | null>(null)
   353→  const [testResults, setTestResults] = useState<Record<string, { ok: boolean; msg: string } | null>>({})
   354→
   355→  // Load channels from API
   356→  useEffect(() => {
   357→    fetch('/api/channels')
   358→      .then(r => r.json())
   359→      .then(data => { if (Array.isArray(data)) setChannels(data) })
   360→      .catch(() => {})
   361→      .finally(() => setLoading(false))
   362→  }, [])
   363→
   364→  const toggleChannel = async (key: string) => {
   365→    const ch = channels.find(c => c.key === key)
   366→    if (!ch) return
   367→    setChannels(prev => prev.map(c => c.key === key ? { ...c, enabled: !c.enabled } : c))
   368→    await fetch('/api/channels', {
   369→      method: 'PUT',
   370→      headers: { 'Content-Type': 'application/json' },
   371→      body: JSON.stringify({ channel: key, enabled: !ch.enabled, config: ch.config }),
   372→    }).catch(() => {})
   373→  }
   374→
   375→  const startEdit = (ch: ChannelData) => {
   376→    setEditingChannel(ch.key)
   377→    setEditForm({ ...ch.config })
   378→    setTestResults(prev => ({ ...prev, [ch.key]: null }))
   379→  }
   380→
   381→  const saveChannel = async (key: string) => {
   382→    setSaving(true)
   383→    try {
   384→      const res = await fetch('/api/channels', {
   385→        method: 'PUT',
   386→        headers: { 'Content-Type': 'application/json' },
   387→        body: JSON.stringify({ channel: key, config: editForm }),
   388→      })
   389→      const data = await res.json()
   390→      setChannels(prev => prev.map(c => c.key === key ? { ...c, configured: data.configured, config: data.config || c.config } : c))
   391→      setEditingChannel(null)
   392→    } catch {}
   393→    setSaving(false)
   394→  }
   395→
   396→  const testConnection = async (key: string) => {
   397→    setTestingKey(key)
   398→    setTestResults(prev => ({ ...prev, [key]: null }))
   399→    try {
   400→      const ch = channels.find(c => c.key === key)
   401→      const res = await fetch('/api/channels/test', {
   402→        method: 'POST',
   403→        headers: { 'Content-Type': 'application/json' },
   404→        body: JSON.stringify({ channel: key, config: { ...ch?.config, ...editForm } }),
   405→      })
   406→      const result = await res.json()
   407→      setTestResults(prev => ({ ...prev, [key]: result }))
   408→      // Update channel data to reflect new test result
   409→      setChannels(prev => prev.map(c => c.key === key ? {
   410→        ...c,
   411→        lastTestAt: new Date().toISOString(),
