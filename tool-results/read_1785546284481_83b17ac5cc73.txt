   503→        <div className="px-3 pt-2 flex gap-2 flex-wrap">
   504→          {attachedFiles.map((file, i) => (
   505→            <div key={i} className="flex items-center gap-1.5 bg-muted/80 rounded-lg px-2.5 py-1.5 text-xs animate-fade-in">
   506→              <Upload className="h-3.5 w-3.5 text-muted-foreground/60" />

   514→        </div>
   515→      )}
   516→
   517→      {/* Message input composer */}
   518→      {convo.status !== 'resolved' && convo.status !== 'closed' && convo.status !== 'spam' ? (
   519→        <div className={cn(
   520→          'composer-area p-3 transition-all duration-300',
   521→          isFocused && 'shadow-[0_-8px_32px_oklch(0.49_0.2_265/0.06)]'
   522→        )}>
   523→          <div className="max-w-3xl mx-auto">
   524→            <div className={cn(
   525→              'flex gap-2 items-end rounded-2xl p-1.5 transition-all duration-300',
   526→              'bg-muted/50 border border-border/40',
   527→              isFocused && 'bg-background border-primary/20 shadow-lg shadow-primary/5'
   528→            )}>
   529→              <div className="flex gap-0.5 pl-1 pb-0.5">
   530→                <TooltipProvider>
   531→                  <Tooltip>
   532→                    <TooltipTrigger asChild>
   533→                      <Button variant="ghost" size="icon" className="h-8 w-8 rounded-xl text-muted-foreground/60 hover:text-muted-foreground hover:bg-foreground/5 transition-all duration-200"

   535→                        <Paperclip className="h-4 w-4" />
   536→                      </Button>
   537→                    </TooltipTrigger>
   538→                    <TooltipContent>Đính kèm tệp</TooltipContent>
   539→                  </Tooltip>
   540→                </TooltipProvider>
   541→                <TooltipProvider>
   542→                  <Tooltip>
   543→                    <TooltipTrigger asChild>
   544→                      <Button variant="ghost" size="icon" className="h-8 w-8 rounded-xl text-muted-foreground/60 hover:text-muted-foreground hover:bg-foreground/5 transition-all duration-200"
   545→                        onClick={() => imageInputRef.current?.click()}>
   546→                        <ImagePlus className="h-4 w-4" />
   547→                      </Button>
   548→                    </TooltipTrigger>
   549→                    <TooltipContent>Gửi hình ảnh</TooltipContent>
   550→                  </Tooltip>
   551→                </TooltipProvider>
   552→                {/* Hidden file inputs */}
   553→                <input ref={fileInputRef} type="file" className="hidden" multiple onChange={handleFileSelect} />
   554→                <input ref={imageInputRef} type="file" className="hidden" accept="image/*" multiple onChange={handleFileSelect} />
   555→              </div>
   556→              <Textarea
   557→                ref={textareaRef}
   558→                value={replyText}
   559→                onChange={(e) => setReplyText(e.target.value)}
   560→                onKeyDown={handleKeyDown}
   561→                onFocus={() => setIsFocused(true)}
   562→                onBlur={() => setIsFocused(false)}
   563→                placeholder="Nhập tin nhắn..."
   564→                className="min-h-[40px] max-h-[120px] resize-none text-[13px] border-0 bg-transparent shadow-none focus-visible:ring-0 focus-visible:border-0 p-2"
   565→                rows={1}
   566→              />
   567→              <div className="flex gap-0.5 pr-0.5 pb-0.5">
   568→                <Popover open={emojiOpen} onOpenChange={setEmojiOpen}>
   569→                  <PopoverTrigger asChild>
   570→                    <Button variant="ghost" size="icon" className="h-8 w-8 rounded-xl text-muted-foreground/60 hover:text-muted-foreground hover:bg-foreground/5 transition-all duration-200">
   571→                      <SmilePlus className="h-4 w-4" />
   572→                    </Button>
   573→                  </PopoverTrigger>
   574→                  <PopoverContent side="top" align="end" className="w-[280px] p-2 rounded-xl">
   575→                    <div className="grid grid-cols-8 gap-0.5">
   576→                      {EMOJI_LIST.map((emoji) => (

   578→                          key={emoji}
   579→                          onClick={() => insertEmoji(emoji)}
   580→                          className="h-8 w-8 flex items-center justify-center rounded-lg text-base hover:bg-foreground/[0.06] transition-colors"
   581→                        >
   582→                          {emoji}
   583→                        </button>
   584→                      ))}
   585→                    </div>
   586→                  </PopoverContent>
   587→                </Popover>
   588→                <Button
   589→                  onClick={handleSend}
   590→                  disabled={(!replyText.trim() && attachedFiles.length === 0) || isSendingMessage}
   591→                  size="icon" className={cn(
   592→                    'h-9 w-9 flex-shrink-0 rounded-xl send-btn',
   593→                    (replyText.trim() || attachedFiles.length > 0) && 'bg-gradient-to-r from-indigo-500 to-violet-500 hover:from-indigo-600 hover:to-violet-600'
   594→                  )}
   595→                >
   596→                  {isSendingMessage ? (
   597→                    <div className="h-4 w-4 border-2 border-white border-t-transparent rounded-full animate-spin" />
   598→                  ) : (
   599→                    <Send className="h-4 w-4" />
   600→                  )}
   601→                </Button>
   602→              </div>
   603→            </div>
   604→            <p className="text-[10px] text-muted-foreground/30 text-center mt-1.5 font-medium">
   605→              Enter để gửi · Shift+Enter xuống dòng
   606→            </p>
   607→          </div>
   608→        </div>
   609→      ) : (
   610→        <div className="composer-area p-4">
   611→          <div className="max-w-3xl mx-auto flex items-center justify-between">
   612→            <span className="text-xs text-muted-foreground/60">
   613→              Hội thoại đã {convo.status === 'resolved' ? 'được giải quyết' : convo.status === 'closed' ? 'đóng' : 'đánh dấu spam'}
   614→            </span>
   615→            <Button variant="outline" size="sm" className="text-xs h-8 rounded-xl font-medium hover:bg-primary hover:text-primary-foreground transition-all duration-200" onClick={() => handleStatusChange('open')}>
   616→              Mở lại
   617→            </Button>
   618→          </div>
   619→        </div>
   620→      )}
   621→    </div>
