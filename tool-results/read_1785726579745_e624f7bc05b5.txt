   698→function SecurityTab() {

   728→  const saveConfig = async () => {

   731→      const res = await fetch('/api/security/config', {
   732→        method: 'PUT',
   733→        headers: { 'Content-Type': 'application/json' },
   734→        body: JSON.stringify({
   735→          ...config,
   736→          rateLimitPerMinute: parseInt(rateLimitInput) || 60,
   737→        }),
   738→      })
   739→      const data = await res.json()
   740→      if (data.data) setConfig(data.data)
   741→    } catch (e) { console.error(e) }
   742→    finally { setSaving(false) }
   743→  }
   744→
   745→  const addToBlacklist = async () => {
   746→    if (!newValue.trim()) return
   747→    setAdding(true)

   749→      const res = await fetch('/api/security/blacklist', {
   750→        method: 'POST',
   751→        headers: { 'Content-Type': 'application/json' },
